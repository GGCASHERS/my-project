local methodUsed = "none"

--========================================================
-- 0. УТИЛИТЫ
--========================================================
local function safeGetInfo(f)
    local ok, info = pcall(debug.getinfo, f)
    if ok then return info end
    return nil
end

local function safeGetUpvalue(f, i)
    local ok, name, val = pcall(debug.getupvalue, f, i)
    if not ok then return nil, nil end
    return name, val
end

local function safeSetUpvalue(f, i, v)
    local ok = pcall(debug.setupvalue, f, i, v)
    return ok
end

-- Собираем все upvalues функции в список {i=, name=, value=}
local function collectUpvalues(f)
    local list = {}
    local i = 1
    while true do
        local name, val = safeGetUpvalue(f, i)
        if name == nil then break end
        list[#list + 1] = { i = i, name = name, value = val }
        i = i + 1
        if i > 300 then break end -- защита от бесконечности
    end
    return list
end

--========================================================
-- 1. ПОИСК ФУНКЦИИ SHOOT
--========================================================
local shootFunc   = nil
local shootInfo   = nil   -- { eqIdx, cdIdx, rlIdx, selfIdx, mouseIdx }
local toolRef     = nil

local function looksLikeShoot(f, info)
    if not info or not info.source then return false end
    if not info.source:find("Revolver") then return false end
    -- shoot обычно короткая функция (не огромный корутин-луп)
    if (info.nups or 0) < 2 then return false end
    return true
end

-- Эвристика: ищем в upvalues таблицу инструмента (self) и mouse
local function analyzeUpvalues(f)
    local ups = collectUpvalues(f)
    local result = { eqIdx = nil, cdIdx = nil, rlIdx = nil, selfIdx = nil, mouseIdx = nil }

    -- Кандидаты на роль «кулдаун» — number upvalue
    local numberIdx = {}
    local boolIdx   = {}

    for _, u in ipairs(ups) do
        local v = u.value
        local t = type(v)

        if t == "table" then
            -- Tool (self) — есть Name и/или Handle
            if v.Name ~= nil or v.Handle ~= nil then
                result.selfIdx = u.i
            end
            -- Mouse — есть Hit и/или Target
            if v.Hit ~= nil or v.Target ~= nil then
                result.mouseIdx = u.i
            end
            -- Иногда self — это {Weapon=..., Character=...}
        elseif t == "number" then
            numberIdx[#numberIdx + 1] = u.i
        elseif t == "boolean" then
            boolIdx[#boolIdx + 1] = u.i
        end
    end

    -- Имена-подсказки (старые и новые варианты)
    for _, u in ipairs(ups) do
        local n = u.name
        if n == "u341" or n == "equipped" or n == "isEquipped" then
            result.eqIdx = u.i
        elseif n == "u344" or n == "cooldown" or n == "lastShot" or n == "lastFire" then
            result.cdIdx = u.i
        elseif n == "u345" or n == "reload" or n == "isReloading" or n == "barrage" then
            result.rlIdx = u.i
        end
    end

    -- Если по именам не нашли — берём первые number/bool
    if not result.cdIdx and #numberIdx > 0 then
        result.cdIdx = numberIdx[1]
    end
    if not result.rlIdx and #boolIdx > 0 then
        result.rlIdx = boolIdx[1]
    end

    return result, ups
end

local function findShoot()
    local okGc, gc = pcall(getgc, true)
    if not okGc or not gc then return false end

    for _, obj in pairs(gc) do
        if type(obj) == "function" then
            local info = safeGetInfo(obj)
            if looksLikeShoot(obj, info) then
                local analysis, ups = analyzeUpvalues(obj)

                -- Должен быть хоть один number (кулдаун) ИЛИ явный self
                if analysis.cdIdx or analysis.selfIdx then
                    shootFunc = obj
                    shootInfo = analysis
                    shootInfo.ups = ups
                    -- Запоминаем toolRef, если нашли self
                    if analysis.selfIdx then
                        toolRef = ups and nil
                        local _, v = safeGetUpvalue(obj, analysis.selfIdx)
                        if type(v) == "table" then toolRef = v end
                    end
                    return true
                end
            end
        end
    end
    return false
end

--========================================================
-- 2. ПАТЧ КУЛДАУНА/ПЕРЕЗАРЯДКИ
--========================================================
local function patchShoot()
    if not shootFunc then
        pcall(findShoot)
        if not shootFunc then return false end
    end
    local info = shootInfo
    if not info then return false end

    if info.eqIdx then safeSetUpvalue(shootFunc, info.eqIdx, true)  end
    if info.cdIdx then safeSetUpvalue(shootFunc, info.cdIdx, 0)     end
    if info.rlIdx then safeSetUpvalue(shootFunc, info.rlIdx, false) end

    -- Дополнительно: если self найден — обнуляем типичные поля Tool
    if toolRef then
        pcall(function() toolRef.Enabled = true end)
        pcall(function() if toolRef:FindFirstChild("Cooldown") then toolRef.Cooldown.Value = 0 end end)
    end

    return true
end

--========================================================
-- 3. ФОРС MOUSE.HIT В ТОЧКУ ПОД КУРСОРОМ
--========================================================
local function getAimPoint()
    -- 1) Пытаемся через mouse.Hit (штатно)
    local ok, hit = pcall(function()
        local m = player:GetMouse()
        return m.Hit
    end)
    if ok and hit then return hit.Position end

    -- 2) Фолбэк: рейкаст из камеры в центр экрана
    local cam = workspace.CurrentCamera
    if cam then
        local ray = Ray.new(cam.CFrame.Position, cam.CFrame.LookVector * 5000)
        local part, pos = workspace:FindPartOnRay(ray, player.Character)
        if pos then return pos end
    end
    return Vector3.new(0, 0, 0)
end

--========================================================
-- 4. ВЫЗОВ ВЫСТРЕЛА (несколько стратегий, первая успешная)
--========================================================
local function strategyCallShoot()
    if not shootFunc then return false end
    local info = shootInfo or {}

    -- Собираем аргументы
    local args = {}

    -- Если первый upvalue-параметр — это self (таблица Tool), передаём его
    -- Lua передаёт self как первый аргумент при вызове через ':'
    -- Мы вызываем через '.', поэтому добавляем self вручную.
    if info.selfIdx then
        local _, selfVal = safeGetUpvalue(shootFunc, info.selfIdx)
        if type(selfVal) == "table" then
            args[1] = selfVal
        end
    end

    -- Патчим и зовём
    patchShoot()
    local ok, err = pcall(function()
        return shootFunc(table.unpack(args))
    end)
    if ok then
        shotCount = shotCount + 1
        lastShotAt = os.clock()
        methodUsed = "shoot()"
        return true
    end
    return false, err
end

local function strategyToolActivate()
    local char = player.Character
    if not char then return false end
    local tool = char:FindFirstChildOfClass("Tool")
    if not tool then return false end
    -- Штатный путь выстрела через движок: Tool:Activate()
    local ok = pcall(function() tool:Activate() end)
    if ok then
        shotCount = shotCount + 1
        lastShotAt = os.clock()
        methodUsed = "Tool:Activate()"
        return true
    end
    return false
end

local function strategyMouseInput()
    local vim = game:GetService("VirtualInputManager")
    if not vim then return false end
    local ok = pcall(function()
        vim:SendMouseButtonEvent(0, 0, 0, true,  game, 1)
        task.wait(0.01)
        vim:SendMouseButtonEvent(0, 0, 0, false, game, 1)
    end)
    if ok then
        shotCount = shotCount + 1
        lastShotAt = os.clock()
        methodUsed = "MouseInput"
        return true
    end
    return false
end

local function fireOnce()
    -- Пробуем по очереди: shoot() -> Tool:Activate() -> MouseInput
    if strategyCallShoot() then return true end
    if strategyToolActivate() then return true end
    if strategyMouseInput() then return true end
    return false
end

--========================================================
-- 5. ХУК OS.CLOCK
--========================================================
local oldClock = os.clock
local fakeTime = oldClock()
local hookSuccess = false

if hookfunction then
    pcall(function()
        hookfunction(os.clock, function()
            if not ENABLED then return oldClock() end
            fakeTime = fakeTime + 0.016
            return fakeTime + 9999
        end)
        hookSuccess = true
    end)
end
if not hookSuccess then
    os.clock = function()
        if not ENABLED then return oldClock() end
        fakeTime = fakeTime + 0.016
        return fakeTime + 9999
    end
end

--========================================================
-- 6. RAPID LOOP
--========================================================
local function startRapid()
    if rapidFire then return end
    rapidFire = true
    task.spawn(function()
        while rapidFire and ENABLED do
            pcall(patchShoot)
            pcall(fireOnce)
            task.wait(fireRate)
        end
    end)
end

local function stopRapid()
    rapidFire = false
end

--========================================================
-- 7. ПЕРЕПОИСК ПРИ ЭКИПИРОВКЕ / РЕСПАВНЕ
--========================================================
local function onCharacter(char)
    task.wait(1.5)
    shootFunc = nil
    shootInfo = nil
    toolRef = nil
    pcall(findShoot)
    char.ChildAdded:Connect(function(child)
        if child.Name == "Revolver" or child:IsA("Tool") then
            task.wait(0.8)
            shootFunc = nil
            shootInfo = nil
            toolRef = nil
            pcall(findShoot)
        end
    end)
end
if player.Character then onCharacter(player.Character) end
player.CharacterAdded:Connect(onCharacter)

-- 9. STATUS LOOP (показывает факт выстрела и метод)
--========================================================
task.spawn(function()
    local lastCount = 0
    while ENABLED do
        task.wait(0.5)
        if not shootFunc then
            pcall(findShoot)
        end
        if shootFunc then
            pcall(patchShoot)
        end

        local now = os.clock()
        local fresh = shotCount - lastCount
        lastCount = shotCount

        if shootFunc then
            status.Text = string.format(
                "● FOUND shoot()  |  %d shots\n   method: %s  |  rate: %.2fs",
                shotCount, methodUsed, fireRate
            )
            status.TextColor3 = fresh > 0 and Color3.fromRGB(70, 255, 110)
                                or Color3.fromRGB(70, 255, 110)
        else
            status.Text = "● SHOOT() NOT FOUND — using fallback\n   method: " .. methodUsed
            status.TextColor3 = Color3.fromRGB(255, 200, 60)
        end
    end
end)
