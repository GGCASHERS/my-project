function New(className, properties)
	local object = Instance.new(className)
	for property, value in pairs(properties) do
		object[property] = value
	end
	return object
end

function Corner(object, radius)
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, radius)
	corner.Parent = object
	return corner
end

function Stroke(object, color, transparency, thickness)
	local stroke = Instance.new("UIStroke")
	stroke.Color = color
	stroke.Transparency = transparency or 0
	stroke.Thickness = thickness or 1
	stroke.Parent = object
	return stroke
end

function Tween(object, duration, properties, style, direction)
	local info = TweenInfo.new(
		duration,
		style or Enum.EasingStyle.Quint,
		direction or Enum.EasingDirection.Out
	)
	local tween = TweenService:Create(object, info, properties)
	tween:Play()
	return tween
end

function keyName(key)
	if not key or key == Enum.KeyCode.Unknown then return "None" end
	local name = key.Name
	name = name:gsub("^Left", "L ")
	name = name:gsub("^Right", "R ")
	return name
end

function inputTypeName(inputType)
	if not inputType then return "None" end
	if inputType == Enum.UserInputType.MouseButton1 then return "LMB"
	elseif inputType == Enum.UserInputType.MouseButton2 then return "RMB"
	elseif inputType == Enum.UserInputType.MouseWheel then return "Wheel"
	elseif inputType == Enum.UserInputType.MouseButton4 then return "MB4"
	elseif inputType == Enum.UserInputType.MouseButton5 then return "MB5"
	elseif inputType == Enum.UserInputType.Keyboard then return "Keyboard"
	else return "Unknown" end
end

function safeDestroy(object)
	if object and object.Parent then
		object:Destroy()
	end
end

local HttpService = game:GetService("HttpService")

local function getExecutorFunction(name)
	local direct = _G[name]
	if type(direct) == "function" then
		return direct
	end
	if type(getgenv) == "function" then
		local env = getgenv()
		if type(env[name]) == "function" then
			return env[name]
		end
	end
	return nil
end

local FS = {
	isfolder = getExecutorFunction("isfolder"),
	makefolder = getExecutorFunction("makefolder"),
	isfile = getExecutorFunction("isfile"),
	writefile = getExecutorFunction("writefile"),
	readfile = getExecutorFunction("readfile"),
	listfiles = getExecutorFunction("listfiles"),
}

local function refreshFS()
	FS.isfolder = getExecutorFunction("isfolder")
	FS.makefolder = getExecutorFunction("makefolder")
	FS.isfile = getExecutorFunction("isfile")
	FS.writefile = getExecutorFunction("writefile")
	FS.readfile = getExecutorFunction("readfile")
	FS.listfiles = getExecutorFunction("listfiles")
end

local function configFileSystemAvailable()
	refreshFS()
	return type(FS.makefolder) == "function"
		and type(FS.isfolder) == "function"
		and type(FS.writefile) == "function"
		and type(FS.readfile) == "function"
		and type(FS.listfiles) == "function"
end

local function ensureConfigFolder()
	refreshFS()
	if not configFileSystemAvailable() then
		return false, "Solara: file API unavailable"
	end

	local folderOK, exists = pcall(FS.isfolder, CONFIG_FOLDER)
	if folderOK and exists == true then
		return true, CONFIG_FOLDER
	end

	local createOK, createErr = pcall(FS.makefolder, CONFIG_FOLDER)
	if not createOK then
		local checkOK, nowExists = pcall(FS.isfolder, CONFIG_FOLDER)
		if not checkOK or not nowExists then
			return false, "Solara: cannot create " .. CONFIG_FOLDER .. ": " .. tostring(createErr)
		end
	end

	local checkOK, nowExists = pcall(FS.isfolder, CONFIG_FOLDER)
	if not checkOK or not nowExists then
		return false, "Solara: folder " .. CONFIG_FOLDER .. " was not created"
	end

	return true, CONFIG_FOLDER
end

local function sanitizeConfigName(name)
	name = tostring(name or "")
	name = name:gsub('[\\/:*?"<>|]', "_")
	name = name:gsub("%s+", " ")
	name = name:match("^%s*(.-)%s*$") or ""
	name = name:gsub("%.json$", "")
	if #name > 48 then
		name = name:sub(1, 48)
	end
	return name
end

local function configPath(name)
	return CONFIG_FOLDER .. "/" .. name .. ".json"
end

local function keyToString(key)
	if not key or key == Enum.KeyCode.Unknown then
		return "Unknown"

	return key.Name
end

local function stringToKey(name)
	if type(name) ~= "string" or name == "Unknown" or name == "" then
		return Enum.KeyCode.Unknown
	end
	return Enum.KeyCode[name] or Enum.KeyCode.Unknown
end

local function saveConfig(configName)
	local folderOK, folderResult = ensureConfigFolder()
	if not folderOK then
		return false, folderResult
	end

	configName = sanitizeConfigName(configName)
	if configName == "" then
		return false, "Enter a config name"
	end

	local encodeOK, encoded = pcall(function()
		return HttpService:JSONEncode(buildConfigData())
	end)
	if not encodeOK then
		return false, "Solara: JSON encode failed: " .. tostring(encoded)
	end

	refreshFS()
	if type(FS.writefile) ~= "function" then
		return false, "Solara: writefile unavailable"
	end

	local path = configPath(configName)
	local writeOK, writeErr = pcall(FS.writefile, path, encoded)
	if not writeOK then
		return false, "Solara: save failed: " .. tostring(writeErr)
	end

	-- Проверяем, что файл реально появился.
	if type(FS.isfile) == "function" then
		local checkOK, exists = pcall(FS.isfile, path)
		if not checkOK or not exists then
			return false, "Solara: file was not created"
		end
	end

	return true, configName
end

local function getConfigFiles()
	local folderOK, folderResult = ensureConfigFolder()
	if not folderOK then
		return {}, folderResult
	end

	refreshFS()
	local listOK, result = pcall(FS.listfiles, CONFIG_FOLDER)
	if not listOK or type(result) ~= "table" then
		return {}, "Solara: listfiles failed"
	end

	local files = {}
	local seen = {}
	for _, path in ipairs(result) do
		local normalized = tostring(path):gsub("\\", "/")
		local name = normalized:match("([^/]+)%.json$")
		if name and not seen[name:lower()] then
			seen[name:lower()] = true
			table.insert(files, {name = name, path = tostring(path)})
		end
	end

	table.sort(files, function(a, b)
		return a.name:lower() < b.name:lower()
	end)
	return files
end

local function loadConfigFile(filePath)
	refreshFS()
	if type(FS.readfile) ~= "function" then
		return false, "Solara: readfile unavailable"
	end

	local readOK, raw = pcall(FS.readfile, filePath)
	if not readOK or type(raw) ~= "string" then
		return false, "Solara: could not read config"
	end

	local decodeOK, data = pcall(function()
		return HttpService:JSONDecode(raw)
	end)
	if not decodeOK or type(data) ~= "table" then
		return false, "Solara: invalid JSON config"
	end

	if type(data.Keys) == "table" then
		for name, value in pairs(data.Keys) do
			if Keys[name] ~= nil then
				Keys[name] = stringToKey(value)
			end
		end
	end

	if type(data.Features) == "table" then
		for name, value in pairs(data.Features) do
			if Features[name] ~= nil then
				Features[name] = value
			end
		end
	end

	if type(data.espSettings) == "table" then
		for name, value in pairs(data.espSettings) do
			if espSettings[name] ~= nil then
				espSettings[name] = value
			end
		end
	end

	if type(espSettings.MaxDistance) ~= "number" then espSettings.MaxDistance = 500 end
	espSettings.MaxDistance = math.clamp(espSettings.MaxDistance, 25, 10000)

	if type(data.Features) == "table" then
		if data.Features.aimbotFOV ~= nil then Features.aimbotFOV = tonumber(data.Features.aimbotFOV) or Features.aimbotFOV end
		if data.Features.aimbotSmoothness ~= nil then Features.aimbotSmoothness = tonumber(data.Features.aimbotSmoothness) or Features.aimbotSmoothness end
		if data.Features.aimbotVisibleCheck ~= nil then Features.aimbotVisibleCheck = data.Features.aimbotVisibleCheck and true or false end
		if data.Features.aimbotTeamCheck ~= nil then Features.aimbotTeamCheck = data.Features.aimbotTeamCheck and true or false end
		if data.Features.aimbotTrackPart ~= nil then Features.aimbotTrackPart = tostring(data.Features.aimbotTrackPart) end
		if data.Features.wallshotMode ~= nil then Features.wallshotMode = tostring(data.Features.wallshotMode) end
		if data.Features.wallshotTransparency ~= nil then Features.wallshotTransparency = math.clamp(tonumber(data.Features.wallshotTransparency) or Features.wallshotTransparency, 0, 1) end
		if type(data.Features.hitboxHeadSize) == "table" then Features.hitboxHeadSize = Vector3.new(tonumber(data.Features.hitboxHeadSize[1]) or 6, tonumber(data.Features.hitboxHeadSize[2]) or 6, tonumber(data.Features.hitboxHeadSize[3]) or 6) end
		if type(data.Features.hitboxBodySize) == "table" then Features.hitboxBodySize = Vector3.new(tonumber(data.Features.hitboxBodySize[1]) or 6, tonumber(data.Features.hitboxBodySize[2]) or 7, tonumber(data.Features.hitboxBodySize[3]) or 6) end
		Features.hitboxTransparency = math.clamp(tonumber(data.Features.hitboxTransparency) or Features.hitboxTransparency, 0, 1)
		Features.freecamDownDistance = math.clamp(tonumber(data.Features.freecamDownDistance) or Features.freecamDownDistance, 10, 1000)
		Features.freecamSpeed = math.max(0.1, tonumber(data.Features.freecamSpeed) or Features.freecamSpeed)
		Features.freecamFastSpeed = math.max(0.1, tonumber(data.Features.freecamFastSpeed) or Features.freecamFastSpeed)
	end

	if data.Features then
		State.keybindsOverlayVisible = data.Features.keybindsOverlayVisible and true or false
		State.activeFeaturesOverlayVisible = data.Features.activeFeaturesOverlayVisible and true or false
	end

	if type(data.theme) == "string" and THEMES[data.theme] then
		currentThemeName = data.theme
	end

	return true, data
end

function makeDraggable(frame, handle)
	handle = handle or frame
	local dragging = false
	local dragInput, dragStart, startPos

	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position

			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	handle.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement then
			dragInput = input
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local delta = input.Position - dragStart
			frame.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset + delta.Y
			)
		end
	end)
end

function createRainbowCursor()
	if rainbowCursorGui then
		rainbowCursorGui:Destroy()
		rainbowCursorGui = nil
		rainbowCursor = nil
	end

	rainbowCursorGui = Instance.new("ScreenGui")
	rainbowCursorGui.Name = "RainbowCursor"
	rainbowCursorGui.Parent = playerGui
	rainbowCursorGui.ResetOnSpawn = false
	rainbowCursorGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

	rainbowCursor = Instance.new("Frame")
	rainbowCursor.Size = UDim2.new(0, 30, 0, 30)
	rainbowCursor.BackgroundTransparency = 0.4
	rainbowCursor.Parent = rainbowCursorGui
	rainbowCursor.ZIndex = 999

	local circle = Instance.new("ImageLabel")
	circle.Size = UDim2.new(1, 0, 1, 0)
	circle.Image = "rbxassetid://3295988371"
	circle.BackgroundTransparency = 1
	circle.Parent = rainbowCursor

	local border = Instance.new("UIStroke")
	border.Color = Color3.new(1, 1, 1)
	border.Thickness = 2
	border.Transparency = 0.3
	border.Parent = rainbowCursor

	Corner(rainbowCursor, 15)

	mouse.Icon = "rbxasset://textures/ui/cursors/blank.cur"

	return rainbowCursor
end

function updateRainbowCursor()
	if not rainbowCursor or not Features.rainbowCursorEnabled then return end
	rainbowHue = (rainbowHue + 0.3) % 360
	local color = Color3.fromHSV(rainbowHue / 360, 1, 1)
	rainbowCursor.BackgroundColor3 = color
	rainbowCursor.Position = UDim2.new(0, mouse.X - 15, 0, mouse.Y - 15)
end

function toggleRainbowCursor(state)
	Features.rainbowCursorEnabled = state
	if state then
		if not rainbowCursorGui then
			createRainbowCursor()
		end
		rainbowCursorGui.Visible = true
		rainbowCursor.Visible = true
		mouse.Icon = "rbxasset://textures/ui/cursors/blank.cur"
	else
		if rainbowCursorGui then
			rainbowCursorGui.Visible = false
		end
		mouse.Icon = ""
	end
end

function createParticle()
	local size = math.random(2, 5)
	local particle = New("Frame", {
		Parent = particleLayer,
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(math.random(5, 95) / 100, math.random(10, 110) / 100),
		Size = UDim2.fromOffset(size, size),
		BackgroundColor3 = COLORS.Accent,
		BackgroundTransparency = math.random(25, 65) / 100,
		BorderSizePixel = 0,
		ZIndex = 3,
	})
	Corner(particle, size)
	table.insert(particleObjects, particle)
	table.insert(themeObjects, particle)
end

for i = 1, 28 do createParticle() end

function animateParticles()
	for _, particle in ipairs(particleObjects) do
		if not particle.Parent then continue end
		local startPosition = particle.Position
		local newX = startPosition.X.Scale + math.random(-5, 5) / 100
		local newY = startPosition.Y.Scale - math.random(5, 15) / 100

		Tween(
			particle,
			math.random(30, 60) / 10,
			{
				Position = UDim2.fromScale(newX, newY),
				BackgroundTransparency = math.random(45, 85) / 100,
			},
			Enum.EasingStyle.Sine
		)
	end
end

task.spawn(function()
	while gui.Parent do
		if particleLayer.Visible then animateParticles() end
		task.wait(2.5)
	end
end)

function makeNavButton(text, icon, y, page)
	local button = New("TextButton", {
		Name = text .. "Button",
		Parent = nav,
		Position = UDim2.fromOffset(0, y),
		Size = UDim2.fromOffset(160, 44),
		BackgroundColor3 = COLORS.Panel,
		BorderSizePixel = 0,
		Text = "",
		AutoButtonColor = false,
		ZIndex = 9,
	})

	Corner(button, 10)

	local indicator = New("Frame", {
		Name = "ActiveIndicator",
		Parent = button,
		Position = UDim2.fromOffset(0, 8),
		Size = UDim2.fromOffset(3, 28),
		BackgroundColor3 = COLORS.Accent,
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ZIndex = 10,
	})

	Corner(indicator, 3)
	table.insert(themeObjects, indicator)

	local iconLabel = New("TextLabel", {
		Name = "NavIcon",
		Parent = button,
		Position = UDim2.fromOffset(17, 0),
		Size = UDim2.fromOffset(25, 44),
		BackgroundTransparency = 1,
		Text = icon,
		TextColor3 = COLORS.SubText,
		TextSize = 15,
		Font = Enum.Font.GothamBold,
		ZIndex = 10,
	})

	local textLabel = New("TextLabel", {
		Name = "NavText",
		Parent = button,
		Position = UDim2.fromOffset(52, 0),
		Size = UDim2.fromOffset(95, 44),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = COLORS.SubText,
		TextSize = 12,
		Font = Enum.Font.GothamBold,
		TextXAlignment = Enum.TextXAlignment.Left,
		ZIndex = 10,
	})

	button.MouseEnter:Connect(function()
		if State.selectedButton ~= button then
			Tween(button, 0.15, { BackgroundColor3 = COLORS.PanelHover })
		end
	end)

	button.MouseLeave:Connect(function()
		if State.selectedButton ~= button then
			Tween(button, 0.15, { BackgroundColor3 = COLORS.Panel })
		end
	end)

	button.MouseButton1Click:Connect(function()
		if State.selectedButton and State.selectedButton ~= button then
			local oldIndicator = State.selectedButton:FindFirstChild("ActiveIndicator")
			local oldIcon = State.selectedButton:FindFirstChild("NavIcon")
			local oldText = State.selectedButton:FindFirstChild("NavText")

			Tween(State.selectedButton, 0.18, { BackgroundColor3 = COLORS.Panel })
			if oldIcon then Tween(oldIcon, 0.18, { TextColor3 = COLORS.SubText }) end
			if oldText then Tween(oldText, 0.18, { TextColor3 = COLORS.SubText }) end
			if oldIndicator then Tween(oldIndicator, 0.18, { BackgroundTransparency = 1 }) end
		end

		State.selectedButton = button
		Tween(button, 0.2, { BackgroundColor3 = Color3.fromRGB(26, 20, 25) })
		Tween(indicator, 0.2, { BackgroundTransparency = 0 })
		Tween(iconLabel, 0.2, { TextColor3 = COLORS.Text })
		Tween(textLabel, 0.2, { TextColor3 = COLORS.Text })

		for _, currentPage in pairs(pages) do
			currentPage.Visible = false
		end

		page.Visible = true
	end)

	return button
end

playerButton = makeNavButton("Player", "●", 0, playerPage)

skinsButton = makeNavButton("Skins", "★", 100, skinsPage)

function updateActiveFeaturesList()
	if not activeFeaturesContainer then return end
	for _, child in ipairs(activeFeaturesContainer:GetChildren()) do
		if child:IsA("TextLabel") then child:Destroy() end
	end
	local enabled = {}
	local function add(name, state) if state then table.insert(enabled, name) end end
	add("Aimbot", Features.aimbotEnabled)
	add("Wallshot", Features.wallshotEnabled)
	add("ESP", Features.espEnabled)
	add("Triggerbot", Features.triggerbotEnabled)
	add("Noclip", Features.noclipEnabled)
	add("Spinbot", Features.spinbotEnabled)
	add("Bunny Hop", Features.bhopEnabled)
	add("Sky Mode", Features.skyEnabled)
	add("Stretch", Features.stretchEnabled)
	add("Anti Fling", Features.antiFlingEnabled)
	add("Anti AFK", Features.antiAFKEnabled)
	add("Rainbow Cursor", Features.rainbowCursorEnabled)
	add("Move Camera", Features.cameraTurnEnabled)
	add("Hitbox", Features.hitboxEnabled)
	add("Invisible / Freecam", Features.freecamEnabled)
	add("Cursor Unlock", Features.cursorUnlocked)
	if #enabled == 0 then
		New("TextLabel", {Parent=activeFeaturesContainer, Size=UDim2.new(1,0,0,18), BackgroundTransparency=1, Text="Nothing enabled", TextColor3=COLORS.Muted, TextSize=9, Font=Enum.Font.GothamMedium, TextXAlignment=Enum.TextXAlignment.Left, ZIndex=101})
	else
		for _, name in ipairs(enabled) do
			New("TextLabel", {Parent=activeFeaturesContainer, Size=UDim2.new(1,0,0,16), BackgroundTransparency=1, Text="●  "..name, TextColor3=COLORS.SubText, TextSize=9, Font=Enum.Font.GothamMedium, TextXAlignment=Enum.TextXAlignment.Left, ZIndex=101})
		end
	end
end

RunService.RenderStepped:Connect(function(deltaTime)
	if Features.spinbotEnabled then
		local character = player.Character
		if not character then return end
		local humanoidRootPart = character:FindFirstChild("HumanoidRootPart")
		local head = character:FindFirstChild("Head")
		if not humanoidRootPart or not head then return end

		local isFirstPerson = (camera.CFrame.Position - head.Position).Magnitude < 1.5

		if isFirstPerson then
			local savedCameraCFrame = camera.CFrame
			humanoidRootPart.CFrame = humanoidRootPart.CFrame * CFrame.Angles(0, math.rad(Features.spinbotSpeed), 0)
			camera.CFrame = savedCameraCFrame
		else
			humanoidRootPart.CFrame = humanoidRootPart.CFrame * CFrame.Angles(0, math.rad(Features.spinbotSpeed), 0)
		end
	end
end)

function getMoveDirection(humanoid)
	local moveDir = humanoid.MoveDirection
	if moveDir.Magnitude == 0 then
		local cam = Workspace.CurrentCamera
		if cam then
			moveDir = cam.CFrame.LookVector
			moveDir = Vector3.new(moveDir.X, 0, moveDir.Z)
		end
	end
	return moveDir.Magnitude > 0 and moveDir.Unit or Vector3.zero
end

RunService.Heartbeat:Connect(function()
	if not Features.bhopEnabled then return end

	if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
		local character = player.Character
		if not character then return end

		local humanoid = character:FindFirstChildOfClass("Humanoid")
		local hrp = character:FindFirstChild("HumanoidRootPart")

		if humanoid and hrp and humanoid.Health > 0 then
			local moveDir = getMoveDirection(humanoid)

			if humanoid.FloorMaterial ~= Enum.Material.Air then
				humanoid.Jump = true

				if Features.bhopMode == "TP" and canTp and moveDir.Magnitude > 0 then
					canTp = false
					local jumpAirTime = 0.35
					local tpDistance = moveDir * (Features.bhopSpeed * jumpAirTime)
					hrp.CFrame = hrp.CFrame + tpDistance
					task.delay(0.15, function()
						canTp = true
					end)
				end
			end

			if Features.bhopMode == "NORMAL" and humanoid.FloorMaterial == Enum.Material.Air then
				if moveDir.Magnitude > 0 then
					local currentY = hrp.AssemblyLinearVelocity.Y
					hrp.AssemblyLinearVelocity = Vector3.new(
						moveDir.X * Features.bhopSpeed,
						currentY,
						moveDir.Z * Features.bhopSpeed
					)
				end
			end
		end
	end
end)

function triggerbotShoot(hit)
	if not hit then return end
	local character = hit.Parent
	if character and character:FindFirstChild("Humanoid") and character.Name ~= player.Name then
		local validPart = false
		
		if Features.triggerTargetPart == "All" then
			validPart = true
		elseif Features.triggerTargetPart == "Head" then
			if hit.Name == "Head" then validPart = true end
		elseif Features.triggerTargetPart == "Torso" then
			if hit.Name == "Torso" or hit.Name == "UpperTorso" or hit.Name == "LowerTorso" or hit.Name == "HumanoidRootPart" then
				validPart = true
			end
		end

		if validPart then
			if Features.triggerMode == "WithAim" then
				if triggerbotFlick then
					triggerbotFlick()
				end
			end
			mouse1press()
			task.wait()
			mouse1release()
		end
	end
end

triggerbotFlick = nil
do
	local flickActive = false
	local flickProgress = 0
	local flickStartCFrame = nil
	local flickTargetCFrame = nil
	local FLICK_DURATION = 0.15
	local FLICK_STRENGTH = 180

	local function getHumanizedOffset(progress)
		local smoothProgress = progress * progress * (3 - 2 * progress)
		local shakeX = math.sin(progress * math.pi * 30) * 0.3 * (1 - progress)
		local shakeY = math.cos(progress * math.pi * 25 + 1.2) * 0.3 * (1 - progress)
		local arcY = math.sin(progress * math.pi) * 1.5
		return smoothProgress, shakeX, shakeY, arcY
	end

	triggerbotFlick = function()
		if flickActive then return end
		if not camera then return end
		
		local randomSign = (math.random(1, 2) == 1 and 1) or -1
		local randomStrength = FLICK_STRENGTH * randomSign * (0.85 + math.random() * 0.3)
		
		local yaw = math.rad(randomStrength)
		local pitch = math.rad(math.random(-5, 5) * 0.5)
		
		local rotationY = CFrame.Angles(0, yaw, 0)
		local rotationX = CFrame.Angles(pitch, 0, 0)
		local totalRotation = rotationX * rotationY
		
		local currentPos = camera.CFrame.Position
		local newCFrame = CFrame.new(currentPos) * (camera.CFrame.Rotation * totalRotation)
		flickTargetCFrame = CFrame.new(currentPos, newCFrame.LookVector + currentPos)
		
		flickStartCFrame = camera.CFrame
		flickProgress = 0
		flickActive = true
	end

	RunService.RenderStepped:Connect(function(deltaTime)
		if not flickActive or not camera then
			if flickActive then flickActive = false end
			return
		end
		
		flickProgress = flickProgress + (deltaTime / FLICK_DURATION)
		
		if flickProgress >= 1 then
			flickProgress = 1
			if flickTargetCFrame then
				camera.CFrame = flickTargetCFrame
			end
			flickActive = false
			return
		end
		
		local smoothP, shakeX, shakeY, arcY = getHumanizedOffset(flickProgress)
		
		local startPos = flickStartCFrame.Position
		local startLook = flickStartCFrame.LookVector
		local startUp = flickStartCFrame.UpVector
		local startRight = flickStartCFrame.RightVector
		
		local targetLook = flickTargetCFrame.LookVector
		local targetUp = flickTargetCFrame.UpVector
		
		local currentLook = startLook:Lerp(targetLook, smoothP)
		local currentUp = startUp:Lerp(targetUp, smoothP)
		
		currentLook = currentLook.Unit
		currentUp = currentUp.Unit
		
		local shakeRight = startRight * math.rad(shakeX)
		local shakeUp = startUp * math.rad(shakeY)
		local shakeOffset = shakeRight + shakeUp
		
		local arcOffset = startUp * math.rad(arcY * 0.8)
		
		local noisyLook = (currentLook + shakeOffset + arcOffset).Unit
		
		local newCFrame = CFrame.lookAt(startPos, startPos + noisyLook, currentUp)
		
		camera.CFrame = newCFrame
	end)
end

RunService.RenderStepped:Connect(function()
	if Features.triggerbotEnabled and mouse.Target then
		local hit = mouse.Target
		triggerbotShoot(hit)
	end
end)

local function getClosestAimbotTarget()
	local closestPlayer, closestPart = nil, nil
	local shortestDist = math.clamp(tonumber(Features.aimbotFOV) or 200, 10, 1000)
	local center = Vector2.new(camera.ViewportSize.X / 2, camera.ViewportSize.Y / 2)

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= player and plr.Character then
			if Features.aimbotTeamCheck and plr.Team == player.Team then
				continue
			end

			local hum = plr.Character:FindFirstChildOfClass("Humanoid")
			local part = plr.Character:FindFirstChild(Features.aimbotTrackPart)
				or plr.Character:FindFirstChild("HumanoidRootPart")

			if hum and hum.Health > 0 and part then
				if Features.aimbotVisibleCheck then
					local rayParams = RaycastParams.new()
					rayParams.FilterDescendantsInstances = {player.Character, camera}
					rayParams.FilterType = Enum.RaycastFilterType.Exclude
					local origin = camera.CFrame.Position
					local direction = (part.Position - origin)
					if direction.Magnitude > 0 then
						local result = Workspace:Raycast(origin, direction, rayParams)
						if result and result.Instance then
							local hitModel = result.Instance:FindFirstAncestorOfClass("Model")
							if hitModel ~= plr.Character then
								continue
							end
						end
					end
				end

				local screenPos, onScreen = camera:WorldToViewportPoint(part.Position)
				if onScreen then
					local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
					if dist < shortestDist then
						shortestDist = dist
						closestPlayer = plr
						closestPart = part
					end
				end
			end
		end
	end

	return closestPlayer, closestPart
end

local wallshotSavedParts = {}
local wallshotActivatedParts = {}
local wallshotGameFolder = Workspace:FindFirstChild("Game")
local wallshotConnections = {}
local wallshotLastCheck = 0

local function wallshotSavePart(part)
	if not part:IsA("BasePart") or wallshotSavedParts[part] then return end
	wallshotSavedParts[part] = {
		CanCollide = part.CanCollide,
		CanQuery = part.CanQuery,
		CanTouch = part.CanTouch,
		Transparency = part.Transparency,
		LocalTransparencyModifier = part.LocalTransparencyModifier,
	}
end

local function wallshotApplyPart(part)
	if not part:IsA("BasePart") then return end
	wallshotSavePart(part)
	pcall(function()
		part.CanCollide = false
		part.CanQuery = false
		part.CanTouch = false
		if Features.wallshotMode == "Transparent" then
			part.Transparency = Features.wallshotTransparency
		else
			local old = wallshotSavedParts[part]
			part.Transparency = old and old.Transparency or part.Transparency
		end
	end)
end

local function wallshotProcessObject(object)
	if object:IsA("BasePart") then wallshotApplyPart(object) end
	for _, d in ipairs(object:GetDescendants()) do
		if d:IsA("BasePart") then wallshotApplyPart(d) end
	end
end

local function wallshotApplyAllVisuals()
	for part, data in pairs(wallshotSavedParts) do
		if part and part.Parent then
			pcall(function()
				if Features.wallshotMode == "Transparent" then
					part.Transparency = Features.wallshotTransparency
				else
					part.Transparency = data.Transparency
					part.LocalTransparencyModifier = data.LocalTransparencyModifier
				end
			end)
		end
	end
end

local function wallshotActivatePart(part)
	if not part or not wallshotSavedParts[part] or wallshotActivatedParts[part] then return end
	wallshotActivatedParts[part] = true
	pcall(function()
		part.CanCollide = true
		part.CanQuery = false
		part.CanTouch = false
		if Features.wallshotMode == "Transparent" then part.Transparency = Features.wallshotTransparency end
	end)
end

local function wallshotFindPartsUnderPlayer()
	local result = {}
	local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
	if not root or not root.Parent or not wallshotGameFolder then return result end

	for part in pairs(wallshotSavedParts) do
		if part and part.Parent then pcall(function() part.CanQuery = true end) end
	end

	local overlapParams = OverlapParams.new()
	overlapParams.FilterType = Enum.RaycastFilterType.Include
	overlapParams.FilterDescendantsInstances = {wallshotGameFolder}
	local parts = Workspace:GetPartBoundsInBox(root.CFrame + Vector3.new(0, -2.5, 0), Vector3.new(5,4,5), overlapParams)
	for _, part in ipairs(parts) do
		if wallshotSavedParts[part] then
			local topY = part.Position.Y + part.Size.Y / 2
			if topY <= root.Position.Y + 1.5 then result[part] = true end
		end
	end

	local rayParams = RaycastParams.new()
	rayParams.FilterType = Enum.RaycastFilterType.Include
	rayParams.FilterDescendantsInstances = {wallshotGameFolder}
	rayParams.IgnoreWater = true
	local center = root.Position
	local offsets = {
		Vector3.new(0,0,0), Vector3.new(1.5,0,0), Vector3.new(-1.5,0,0),
		Vector3.new(0,0,1.5), Vector3.new(0,0,-1.5), Vector3.new(1.2,0,1.2),
		Vector3.new(-1.2,0,1.2), Vector3.new(1.2,0,-1.2), Vector3.new(-1.2,0,-1.2)
	}
	for _, offset in ipairs(offsets) do
		local ray = Workspace:Raycast(center + offset + Vector3.new(0,1,0), Vector3.new(0,-6,0), rayParams)
		if ray and wallshotSavedParts[ray.Instance] then result[ray.Instance] = true end
	end

	for part in pairs(wallshotSavedParts) do
		if part and part.Parent then pcall(function() part.CanQuery = false end) end
	end
	return result
end

function setWallshot(state)
	Features.wallshotEnabled = state
	if Features.wallshotEnabled then
		wallshotGameFolder = wallshotGameFolder or Workspace:FindFirstChild("Game")
		if wallshotGameFolder then
			if next(wallshotSavedParts) == nil then
				for _, object in ipairs(wallshotGameFolder:GetChildren()) do wallshotProcessObject(object) end
			end
			wallshotApplyAllVisuals()
		end
		Tween(wallshotToggle, 0.2, {BackgroundColor3 = COLORS.Accent})
		Tween(wallshotKnob, 0.2, {Position = UDim2.fromOffset(24,3), BackgroundColor3 = Color3.new(1,1,1)})
	else
		Tween(wallshotToggle, 0.2, {BackgroundColor3 = COLORS.Panel2})
		Tween(wallshotKnob, 0.2, {Position = UDim2.fromOffset(3,3), BackgroundColor3 = COLORS.SubText})
		for part, data in pairs(wallshotSavedParts) do
			if part and part.Parent then pcall(function()
				part.CanCollide = data.CanCollide
				part.CanQuery = data.CanQuery
				part.CanTouch = data.CanTouch
				part.Transparency = data.Transparency
				part.LocalTransparencyModifier = data.LocalTransparencyModifier
			end) end
		end
		wallshotSavedParts = {}
		wallshotActivatedParts = {}
	end
end

local function hitboxCreate(part, size, color)

end

local freecamEnabled = false
local freecamSavedCharacterCFrame = nil
local freecamCameraCFrame = nil
local freecamConnection = nil
local freecamInputBeganConnection = nil
local freecamInputEndedConnection = nil
local freecamYaw = 0
local freecamPitch = 0
local freecamKeys = {W=false,A=false,S=false,D=false,Q=false,E=false,Shift=false}

local function freecamGetCharacter()
	local character = player.Character or player.CharacterAdded:Wait()
	local root = character:WaitForChild("HumanoidRootPart")
	local humanoid = character:WaitForChild("Humanoid")
	return character, root, humanoid
end
local function freecamKeyDown(input)
	if input.KeyCode == Enum.KeyCode.W then freecamKeys.W=true elseif input.KeyCode == Enum.KeyCode.A then freecamKeys.A=true elseif input.KeyCode == Enum.KeyCode.S then freecamKeys.S=true elseif input.KeyCode == Enum.KeyCode.D then freecamKeys.D=true elseif input.KeyCode == Enum.KeyCode.Q then freecamKeys.Q=true elseif input.KeyCode == Enum.KeyCode.E then freecamKeys.E=true elseif input.KeyCode == Enum.KeyCode.LeftShift then freecamKeys.Shift=true end
end
local function freecamKeyUp(input)
	if input.KeyCode == Enum.KeyCode.W then freecamKeys.W=false elseif input.KeyCode == Enum.KeyCode.A then freecamKeys.A=false elseif input.KeyCode == Enum.KeyCode.S then freecamKeys.S=false elseif input.KeyCode == Enum.KeyCode.D then freecamKeys.D=false elseif input.KeyCode == Enum.KeyCode.Q then freecamKeys.Q=false elseif input.KeyCode == Enum.KeyCode.E then freecamKeys.E=false elseif input.KeyCode == Enum.KeyCode.LeftShift then freecamKeys.Shift=false end
end
local function enableFreecam()
	if freecamEnabled then return end
	local character, root, humanoid = freecamGetCharacter()
	freecamEnabled=true
	Features.freecamEnabled=true
	freecamSavedCharacterCFrame=root.CFrame
	freecamCameraCFrame=camera.CFrame
	local look=freecamCameraCFrame.LookVector
	freecamYaw=math.atan2(-look.X,-look.Z)
	freecamPitch=math.asin(math.clamp(look.Y,-1,1))
	humanoid.AutoRotate=false
	root.CFrame=freecamSavedCharacterCFrame*CFrame.new(0,-Features.freecamDownDistance,0)
	root.Anchored=true
	root.AssemblyLinearVelocity=Vector3.zero
	root.AssemblyAngularVelocity=Vector3.zero
	camera.CameraType=Enum.CameraType.Scriptable
	camera.CFrame=freecamCameraCFrame
	UserInputService.MouseBehavior=Enum.MouseBehavior.LockCenter
	UserInputService.MouseIconEnabled=false
	freecamConnection=RunService.RenderStepped:Connect(function(deltaTime)
		if not freecamEnabled then return end
		if not root or not root.Parent then return end
		local mouseDelta=UserInputService:GetMouseDelta()
		freecamYaw-=mouseDelta.X*0.0025
		freecamPitch=math.clamp(freecamPitch-mouseDelta.Y*0.0025,math.rad(-89),math.rad(89))
		local rotation=CFrame.Angles(0,freecamYaw,0)*CFrame.Angles(freecamPitch,0,0)
		local move=Vector3.zero
		if freecamKeys.W then move+=Vector3.new(0,0,-1) end
		if freecamKeys.S then move+=Vector3.new(0,0,1) end
		if freecamKeys.A then move+=Vector3.new(-1,0,0) end
		if freecamKeys.D then move+=Vector3.new(1,0,0) end
		if freecamKeys.E then move+=Vector3.new(0,1,0) end
		if freecamKeys.Q then move+=Vector3.new(0,-1,0) end
		if move.Magnitude>0 then local speed=freecamKeys.Shift and Features.freecamFastSpeed or Features.freecamSpeed; freecamCameraCFrame+=rotation:VectorToWorldSpace(move.Unit)*speed*60*deltaTime end
		camera.CFrame=CFrame.new(freecamCameraCFrame.Position)*rotation
		root.CFrame=freecamSavedCharacterCFrame*CFrame.new(0,-Features.freecamDownDistance,0)
		root.AssemblyLinearVelocity=Vector3.zero
		root.AssemblyAngularVelocity=Vector3.zero
	end)
	freecamInputBeganConnection=UserInputService.InputBegan:Connect(freecamKeyDown)
	freecamInputEndedConnection=UserInputService.InputEnded:Connect(freecamKeyUp)
end
local function disableFreecam()
	if not freecamEnabled then return end
	local currentCameraCFrame=camera.CFrame
	local currentCameraPosition=currentCameraCFrame.Position
	local currentCameraLook=currentCameraCFrame.LookVector
	freecamEnabled=false
	Features.freecamEnabled=false
	if freecamConnection then freecamConnection:Disconnect(); freecamConnection=nil end
	if freecamInputBeganConnection then freecamInputBeganConnection:Disconnect(); freecamInputBeganConnection=nil end
	if freecamInputEndedConnection then freecamInputEndedConnection:Disconnect(); freecamInputEndedConnection=nil end
	local character=player.Character
	local root=character and character:FindFirstChild("HumanoidRootPart")
	local humanoid=character and character:FindFirstChildOfClass("Humanoid")
	if root then root.Anchored=false; root.CFrame=CFrame.lookAt(currentCameraPosition+Vector3.new(0,2,0),currentCameraPosition+Vector3.new(0,2,0)+currentCameraLook); root.AssemblyLinearVelocity=Vector3.zero; root.AssemblyAngularVelocity=Vector3.zero end
	if humanoid then humanoid.AutoRotate=true end
	camera.CameraType=Enum.CameraType.Custom
	if humanoid then camera.CameraSubject=humanoid end
	UserInputService.MouseBehavior=Enum.MouseBehavior.Default
	UserInputService.MouseIconEnabled=true
	freecamSavedCharacterCFrame=nil
	freecamCameraCFrame=nil
	for k in pairs(freecamKeys) do freecamKeys[k]=false end
end
function setFreecam(state)
	Features.freecamEnabled=state
	if state then
		if not freecamEnabled then enableFreecam() end
		Tween(freecamToggle,0.2,{BackgroundColor3=COLORS.Accent})
		Tween(freecamKnob,0.2,{Position=UDim2.fromOffset(24,3),BackgroundColor3=Color3.new(1,1,1)})

		Tween(freecamToggle,0.2,{BackgroundColor3=COLORS.Panel2})
		Tween(freecamKnob,0.2,{Position=UDim2.fromOffset(3,3),BackgroundColor3=COLORS.SubText})
	end
end

function startCameraTurn()
	local camera = workspace.CurrentCamera
	if not camera then return end

	local ACTIVATION_KEY = Enum.UserInputType.MouseButton1
	local TURN_DURATION = 0.15
	local TURN_STRENGTH = 180

	local isTurning = false
	local turnProgress = 0
	local startCFrame = nil
	local targetCFrame = nil

	local function getHumanizedOffset(progress)
		local smoothProgress = progress * progress * (3 - 2 * progress)
		local shakeX = math.sin(progress * math.pi * 30) * 0.3 * (1 - progress)
		local shakeY = math.cos(progress * math.pi * 25 + 1.2) * 0.3 * (1 - progress)
		local arcY = math.sin(progress * math.pi) * 1.5
		return smoothProgress, shakeX, shakeY, arcY
	end

	local function startTurn()
		if isTurning then return end
		if not camera then return end

		local randomSign = (math.random(1, 2) == 1 and 1) or -1
		local randomStrength = TURN_STRENGTH * randomSign * (0.85 + math.random() * 0.3)
		local currentLookVector = camera.CFrame.LookVector
		local currentUpVector = camera.CFrame.UpVector
		local yaw = math.rad(randomStrength)
		local pitch = math.rad(math.random(-5, 5) * 0.5)
		local rotationY = CFrame.Angles(0, yaw, 0)
		local rotationX = CFrame.Angles(pitch, 0, 0)
		local totalRotation = rotationX * rotationY
		local currentPos = camera.CFrame.Position
		local newCFrame = CFrame.new(currentPos) * (camera.CFrame.Rotation * totalRotation)
		targetCFrame = CFrame.new(currentPos, newCFrame.LookVector + currentPos)
		startCFrame = camera.CFrame
		turnProgress = 0
		isTurning = true
	end

	startTurn()

	local connection
	connection = RunService.RenderStepped:Connect(function(deltaTime)
		if not isTurning or not camera then
			if connection then connection:Disconnect() end
			if isTurning then isTurning = false end
			return
		end

		turnProgress = turnProgress + (deltaTime / TURN_DURATION)

		if turnProgress >= 1 then
			turnProgress = 1
			if targetCFrame then
				camera.CFrame = targetCFrame
			end
			isTurning = false
			connection:Disconnect()
			return
		end

		local smoothP, shakeX, shakeY, arcY = getHumanizedOffset(turnProgress)
		local startPos = startCFrame.Position
		local startLook = startCFrame.LookVector
		local startUp = startCFrame.UpVector
		local startRight = startCFrame.RightVector
		local targetLook = targetCFrame.LookVector
		local targetUp = targetCFrame.UpVector
		local currentLook = startLook:Lerp(targetLook, smoothP)
		local currentUp = startUp:Lerp(targetUp, smoothP)
		currentLook = currentLook.Unit
		currentUp = currentUp.Unit
		local shakeRight = startRight * math.rad(shakeX)
		local shakeUp = startUp * math.rad(shakeY)
		local shakeOffset = shakeRight + shakeUp
		local arcOffset = startUp * math.rad(arcY * 0.8)
		local noisyLook = (currentLook + shakeOffset + arcOffset).Unit
		local newCFrame = CFrame.lookAt(startPos, startPos + noisyLook, currentUp)
		camera.CFrame = newCFrame
	end)
end

function applySkyColor(color)
	Lighting.OutdoorAmbient = color
	Lighting.Ambient = color
	Lighting.FogColor = color
	Lighting.FogStart = 0
	Lighting.FogEnd = 2000
	if skyAtmosphere then
		skyAtmosphere.Color = color
	end
end

local lightingSnapshot = nil
local function saveLightingSnapshot()
	if lightingSnapshot then return end
	lightingSnapshot={props={},children={}}
	local props={"Ambient","OutdoorAmbient","FogColor","FogStart","FogEnd","Brightness","ClockTime","GeographicLatitude","ExposureCompensation","EnvironmentDiffuseScale","EnvironmentSpecularScale","ColorShift_Bottom","ColorShift_Top","GlobalShadows","Technology"}
	for _,name in ipairs(props) do local ok,value=pcall(function() return Lighting[name] end); if ok then lightingSnapshot.props[name]=value end end
	for _,child in ipairs(Lighting:GetChildren()) do local ok,clone=pcall(function() return child:Clone() end); if ok and clone then table.insert(lightingSnapshot.children,clone) end end
end
local function restoreLightingSnapshot()
	if not lightingSnapshot then return end
	if skyConnection then skyConnection:Disconnect(); skyConnection=nil end
	for _,child in ipairs(Lighting:GetChildren()) do pcall(function() child:Destroy() end) end
	for _,child in ipairs(lightingSnapshot.children) do pcall(function() local clone=child:Clone(); clone.Parent=Lighting end) end
	for name,value in pairs(lightingSnapshot.props) do pcall(function() Lighting[name]=value end) end
	skyAtmosphere=nil
	lightingSnapshot=nil
end

function startSkyMode()
	saveLightingSnapshot()
	for _, v in pairs(Lighting:GetChildren()) do
		if v:IsA("Sky") then
			v:Destroy()
		end
	end

	skyAtmosphere = Lighting:FindFirstChildOfClass("Atmosphere")
	if not skyAtmosphere then
		skyAtmosphere = Instance.new("Atmosphere")
		skyAtmosphere.Parent = Lighting
	end
	skyAtmosphere.Density = 0.6
	skyAtmosphere.Haze = 10
	skyAtmosphere.Glare = 10
	skyAtmosphere.Decay = Color3.fromRGB(0, 0, 0)

	if Features.skyMode == "Rainbow" then
		skyConnection = RunService.RenderStepped:Connect(function()
			local hue = (tick() % 2) / 2
			local rainbowColor = Color3.fromHSV(hue, 1, 1)
			applySkyColor(rainbowColor)
		end)
	elseif Features.skyMode == "Red" then
		applySkyColor(Color3.fromRGB(255, 0, 0))
	elseif Features.skyMode == "Blue" then
		applySkyColor(Color3.fromRGB(0, 150, 255))
	end
end

function stopSkyMode()
	if lightingSnapshot then restoreLightingSnapshot() elseif skyConnection then skyConnection:Disconnect(); skyConnection=nil end
end

function startStretch()
	getgenv().Resolution = {
		[".gg/scripters"] = 0.65
	}
	if getgenv().gg_scripters == nil then
		stretchConnection = RunService.RenderStepped:Connect(function()
			camera.CFrame = camera.CFrame * CFrame.new(0, 0, 0, 1, 0, 0, 0, getgenv().Resolution[".gg/scripters"], 0, 0, 0, 1)
		end)
	end
	getgenv().gg_scripters = "GGCASHERS"
end

function stopStretch()
	if stretchConnection then
		stretchConnection:Disconnect()
		stretchConnection = nil
	end
	getgenv().Resolution = nil
	getgenv().gg_scripters = nil
end

function createESPForPlayer(plr)
	if plr == player then return end

	local box = New("Frame", {
		Parent = espLayer,
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		Visible = false
	})
	Stroke(box, COLORS.Accent, 0, 1.5)

	local tracer = New("Frame", {
		Parent = espLayer,
		BackgroundColor3 = COLORS.Accent,
		BorderSizePixel = 0,
		AnchorPoint = Vector2.new(0.5, 0.5),
		Visible = false
	})

	local nameTag = New("TextLabel", {
		Parent = espLayer,
		BackgroundTransparency = 1,
		TextColor3 = Color3.new(1, 1, 1),
		TextSize = 11,
		Font = Enum.Font.GothamBold,
		Visible = false,
		TextStrokeTransparency = 0.3
	})

	local toolTag = New("TextLabel", {
		Parent = espLayer,
		BackgroundTransparency = 1,
		TextColor3 = COLORS.SubText,
		TextSize = 9,
		Font = Enum.Font.GothamMedium,
		Visible = false,
		TextStrokeTransparency = 0.3
	})

	local hpBg = New("Frame", {
		Parent = espLayer,
		BackgroundColor3 = Color3.fromRGB(20, 20, 20),
		BorderSizePixel = 0,
		Visible = false
	})

	local hpBar = New("Frame", {
		Parent = hpBg,
		BackgroundColor3 = Color3.fromRGB(50, 220, 50),
		BorderSizePixel = 0,
		Size = UDim2.fromScale(1, 1)
	})

	espDrawings[plr] = {
		box = box,
		tracer = tracer,
		nameTag = nameTag,
		toolTag = toolTag,
		hpBg = hpBg,
		hpBar = hpBar
	}
end

function removeESP(plr)
	if espDrawings[plr] then
		for _, v in pairs(espDrawings[plr]) do
			safeDestroy(v)
		end
		espDrawings[plr] = nil
	end
end

for _, p in ipairs(Players:GetPlayers()) do createESPForPlayer(p) end
Players.PlayerAdded:Connect(createESPForPlayer)
Players.PlayerRemoving:Connect(removeESP)

RunService.RenderStepped:Connect(function()
	for plr, draw in pairs(espDrawings) do
		local char = plr.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local root = char and char:FindFirstChild("HumanoidRootPart")

		if Features.espEnabled and char and hum and root and hum.Health > 0 then
			local localRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
			local worldDistance = localRoot and (localRoot.Position - root.Position).Magnitude or math.huge
			local withinDistance = worldDistance <= (tonumber(espSettings.MaxDistance) or 500)
			local rootPos, onScreen = camera:WorldToViewportPoint(root.Position)
			if not withinDistance then onScreen = false end
			if onScreen then
				local head = char:FindFirstChild("Head")
				local headPos = head and camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0)) or rootPos

				local sizeY = math.abs(headPos.Y - rootPos.Y) * 2
				sizeY = math.clamp(sizeY, 15, 500)
				local sizeX = sizeY * 0.65

				if espSettings.Boxes then
					draw.box.Size = UDim2.fromOffset(sizeX, sizeY)
					draw.box.Position = UDim2.fromOffset(rootPos.X - sizeX/2, rootPos.Y - sizeY/2)
					draw.box.Visible = true
				else
					draw.box.Visible = false
				end

				if espSettings.Tracers then
					local origin = Vector2.new(camera.ViewportSize.X / 2, camera.ViewportSize.Y)
					local target = Vector2.new(rootPos.X, rootPos.Y)
					local dist = (target - origin).Magnitude

					draw.tracer.Size = UDim2.fromOffset(1.5, dist)
					draw.tracer.Position = UDim2.fromOffset((origin.X + target.X)/2, (origin.Y + target.Y)/2)
					draw.tracer.Rotation = math.deg(math.atan2(target.Y - origin.Y, target.X - origin.X)) - 90
					draw.tracer.Visible = true
				else
					draw.tracer.Visible = false
				end

				if espSettings.ShowName then
					local text = plr.Name
					if espSettings.ShowDistance then
						local distMeters = math.floor((player.Character and player.Character:FindFirstChild("HumanoidRootPart") and (player.Character.HumanoidRootPart.Position - root.Position).Magnitude) or 0)
						text = text .. " [" .. distMeters .. "m]"
					end
					draw.nameTag.Text = text
					draw.nameTag.Position = UDim2.fromOffset(rootPos.X - 100, rootPos.Y - sizeY/2 - 18)
					draw.nameTag.Size = UDim2.fromOffset(200, 15)
					draw.nameTag.Visible = true
				else
					draw.nameTag.Visible = false
				end

				if espSettings.ShowHeldTool then
					local tool = char:FindFirstChildOfClass("Tool")
					draw.toolTag.Text = tool and tool.Name or "None"
					draw.toolTag.Position = UDim2.fromOffset(rootPos.X - 100, rootPos.Y + sizeY/2 + 2)
					draw.toolTag.Size = UDim2.fromOffset(200, 15)
					draw.toolTag.Visible = true
				else
					draw.toolTag.Visible = false
				end

				if espSettings.HealthBar then
					local healthPercent = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
					draw.hpBg.Size = UDim2.fromOffset(3, sizeY)
					draw.hpBg.Position = UDim2.fromOffset(rootPos.X - sizeX/2 - 6, rootPos.Y - sizeY/2)
					draw.hpBar.Size = UDim2.new(1, 0, healthPercent, 0)
					draw.hpBar.Position = UDim2.new(0, 0, 1 - healthPercent, 0)
					draw.hpBar.BackgroundColor3 = Color3.fromRGB(255 - (255 * healthPercent), 255 * healthPercent, 0)
					draw.hpBg.Visible = true
				else
					draw.hpBg.Visible = false
				end

			else
				draw.box.Visible = false
				draw.tracer.Visible = false
				draw.nameTag.Visible = false
				draw.toolTag.Visible = false
				draw.hpBg.Visible = false
			end
		else
			draw.box.Visible = false
			draw.tracer.Visible = false
			draw.nameTag.Visible = false
			draw.toolTag.Visible = false
			draw.hpBg.Visible = false
		end
	end
end)

function refreshConfigList()
	for _, child in ipairs(configListScroll:GetChildren()) do
		if child:IsA("TextButton") then child:Destroy() end
	end

	local files, err = getConfigFiles()
	if err then
		local row = New("TextLabel", {
			Parent = configListScroll,
			Size = UDim2.new(1, -8, 0, 34),
			BackgroundTransparency = 1,
			Text = err,
			TextColor3 = COLORS.SubText,
			TextSize = 10,
			Font = Enum.Font.GothamMedium,
			TextWrapped = true,
		})
		return
	end

	if #files == 0 then
		New("TextLabel", {
			Parent = configListScroll,
			Size = UDim2.new(1, -8, 0, 34),
			BackgroundTransparency = 1,
			Text = "No configs in " .. CONFIG_FOLDER,
			TextColor3 = COLORS.SubText,
			TextSize = 10,
			Font = Enum.Font.GothamMedium,
		})
	else
		for _, cfg in ipairs(files) do
			local btn = New("TextButton", {
				Parent = configListScroll,
				Size = UDim2.new(1, -8, 0, 34),
				BackgroundColor3 = COLORS.Panel2,
				BorderSizePixel = 0,
				Text = cfg.name,
				TextColor3 = COLORS.Text,
				TextSize = 10,
				Font = Enum.Font.GothamBold,
				TextXAlignment = Enum.TextXAlignment.Left,
				AutoButtonColor = false,
			})
			Corner(btn, 8)
			btn.MouseButton1Click:Connect(function()
				local ok, result = loadConfigFile(cfg.path)
				if not ok then
					settingsStatus.Text = result
					return
				end

				-- Применяем загруженные состояния к UI.
				setTriggerbot(Features.triggerbotEnabled)
				setCursorUnlock(Features.cursorUnlocked)
				setNoclip(Features.noclipEnabled)
				setESP(Features.espEnabled)
				setSpinbot(Features.spinbotEnabled)
				setBhop(Features.bhopEnabled)
				setSky(Features.skyEnabled)
				setStretch(Features.stretchEnabled)
				setAimbot(Features.aimbotEnabled)
				setWallshot(Features.wallshotEnabled)
				setHitbox(Features.hitboxEnabled)
				if Features.freecamEnabled then setFreecam(true) else setFreecam(false) end
				if aimbotFOVBox then aimbotFOVBox.Text = tostring(Features.aimbotFOV) end
				if aimbotSmoothBox then aimbotSmoothBox.Text = tostring(Features.aimbotSmoothness) end
				if aimbotTrackBtn then aimbotTrackBtn.Text = Features.aimbotTrackPart end
				if wallshotModeBtn then wallshotModeBtn.Text = Features.wallshotMode end
				if wallshotTransparencyBox then wallshotTransparencyBox.Text = tostring(Features.wallshotTransparency) end
				if toggleAimbotFOVVisual then toggleAimbotFOVVisual() end

--// =========================================================

function startAntiFling()
	local lastPosition = nil
	
	local steppedConn = RunService.Stepped:Connect(function()
		for _, plr in ipairs(Players:GetPlayers()) do
			if plr ~= player and plr.Character then
				local char = plr.Character
				for _, part in ipairs(char:GetDescendants()) do
					if part:IsA("BasePart") then
						part.CanCollide = false
						part.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
						part.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
						part.CustomPhysicalProperties = PhysicalProperties.new(0, 0, 0, 0, 0)
					end
				end
			end
		end
	end)
	table.insert(antiFlingConnections, steppedConn)

	local heartbeatConn = RunService.Heartbeat:Connect(function()
		pcall(function()
			local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
			if hrp then
				if hrp.AssemblyLinearVelocity.Magnitude > 200 or hrp.AssemblyAngularVelocity.Magnitude > 200 then
					hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
					hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
					if lastPosition then
						hrp.CFrame = lastPosition
					end
				elseif hrp.AssemblyLinearVelocity.Magnitude < 50 then
					lastPosition = hrp.CFrame
				end
			end
		end)
	end)
	table.insert(antiFlingConnections, heartbeatConn)
end

function stopAntiFling()
	for _, conn in ipairs(antiFlingConnections) do
		conn:Disconnect()
	end
	antiFlingConnections = {}
end

function applyTheme(color)
	COLORS.Accent = color
	openStroke.Color = color
	if Features.triggerbotEnabled then tbToggle.BackgroundColor3 = color end
	if Features.cursorUnlocked then cursorToggle.BackgroundColor3 = color end
	if Features.noclipEnabled then noclipToggle.BackgroundColor3 = color end
	if Features.espEnabled then espToggle.BackgroundColor3 = color end
	if Features.spinbotEnabled then spinbotToggle.BackgroundColor3 = color end
	if Features.bhopEnabled then bhopToggle.BackgroundColor3 = color end
	if Features.skyEnabled then skyToggle.BackgroundColor3 = color end
	if Features.stretchEnabled then stretchToggle.BackgroundColor3 = color end
	espRebindBtn.TextColor3 = color
	for _, obj in ipairs(themeObjects) do
		if obj:IsA("Frame") then
			if obj.Name == "ActiveIndicator" and obj.BackgroundTransparency == 1 then
				obj.BackgroundColor3 = color
			else
				obj.BackgroundColor3 = color
			end

function openMenu()
	if State.opened or State.animating then return end
	State.animating = true
	State.opened = true
	particleLayer.Visible = true
	main.Visible = true
	openButton.Visible = false
	main.Size = CLOSED_SIZE

	Tween(main, 0.45, { Size = MENU_SIZE }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
	task.delay(0.45, function() if State.opened then State.animating = false end end)
end

function closeMenu()
	if not State.opened or State.animating then return end
	State.animating = true
	State.opened = false

	Tween(main, 0.3, { Size = CLOSED_SIZE }, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
	task.delay(0.3, function()
		if not State.opened then
			main.Visible = false
			particleLayer.Visible = false
			openButton.Visible = true
		end
		State.animating = false
	end)
end

function toggleMenu()
	if State.opened then closeMenu() else openMenu() end
end

function openSettings()
	if State.settingsOpen then return end
	State.settingsOpen = true
	settingsPopup.Visible = true
	settingsPopup.Size = UDim2.fromOffset(0, 0)
	Tween(settingsPopup, 0.35, { Size = SETTINGS_SIZE }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
end

function closeSettingsMenu()
	if not State.settingsOpen then return end
	State.settingsOpen = false
	Tween(settingsPopup, 0.25, { Size = UDim2.fromOffset(0, 0) })
	task.delay(0.25, function()
		if not State.settingsOpen and settingsPopup.Parent then settingsPopup.Visible = false end
	end)
end

settingsButton.MouseButton1Click:Connect(function()
	if State.settingsOpen then closeSettingsMenu() else openSettings() end
end)

closeSettings.MouseButton1Click:Connect(closeSettingsMenu)
openButton.MouseButton1Click:Connect(toggleMenu)

local respawnPersistenceConnection = player.CharacterAdded:Connect(function(character)
	task.wait(0.25)
	if Features.hitboxEnabled then hitboxSetupCharacter(character) end
	if Features.freecamEnabled then
		freecamEnabled=false
		if freecamConnection then freecamConnection:Disconnect(); freecamConnection=nil end
		if freecamInputBeganConnection then freecamInputBeganConnection:Disconnect(); freecamInputBeganConnection=nil end
		if freecamInputEndedConnection then freecamInputEndedConnection:Disconnect(); freecamInputEndedConnection=nil end
		camera=Workspace.CurrentCamera
		task.wait(0.4)
		if Features.freecamEnabled then pcall(function() enableFreecam() end) end
	end
	if Features.noclipEnabled then pcall(function() setNoclip(true) end) end
	if Features.antiFlingEnabled then pcall(function() stopAntiFling(); startAntiFling() end) end
end)

function unloadUI()
	setAimbot(false)
	setWallshot(false)
	setHitbox(false)
	setFreecam(false)
	for _, connection in pairs(hitboxConnections) do pcall(function() connection:Disconnect() end) end
	hitboxConnections = {}
	if respawnPersistenceConnection then pcall(function() respawnPersistenceConnection:Disconnect() end); respawnPersistenceConnection=nil end
	for _, connection in ipairs(wallshotConnections) do pcall(function() connection:Disconnect() end) end
	stopSkyMode()
	stopStretch()
	stopAntiFling()
	stopAntiAFK()
	toggleRainbowCursor(false)
	for _, draw in pairs(espDrawings) do
		for _, v in pairs(draw) do safeDestroy(v) end
	end
	safeDestroy(gui)
	print("ATEOT HUB UNLOADED")
end

unloadButton.MouseButton1Click:Connect(function()
	settingsStatus.Text = "Unloading UI..."
	task.wait(0.3)
	unloadUI()
end)
