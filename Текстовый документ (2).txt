function New(className, properties)
	local object = Instance.new(className)
	for property, value in pairs(properties) do
		object[property] = value
	end
	return object
end

function Corner(object, radius)
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, radius)
	corner.Parent = object
	return corner
end

function Stroke(object, color, transparency, thickness)
	local stroke = Instance.new("UIStroke")
	stroke.Color = color
	stroke.Transparency = transparency or 0
	stroke.Thickness = thickness or 1
	stroke.Parent = object
	return stroke
end

function Tween(object, duration, properties, style, direction)
	local info = TweenInfo.new(
		duration,
		style or Enum.EasingStyle.Quint,
		direction or Enum.EasingDirection.Out
	)
	local tween = TweenService:Create(object, info, properties)
	tween:Play()
	return tween
end

function keyName(key)
	if not key or key == Enum.KeyCode.Unknown then return "None" end
	local name = key.Name
	name = name:gsub("^Left", "L ")
	name = name:gsub("^Right", "R ")
	return name
end

function inputTypeName(inputType)
	if not inputType then return "None" end
	if inputType == Enum.UserInputType.MouseButton1 then return "LMB"
	elseif inputType == Enum.UserInputType.MouseButton2 then return "RMB"
	elseif inputType == Enum.UserInputType.MouseWheel then return "Wheel"
	elseif inputType == Enum.UserInputType.MouseButton4 then return "MB4"
	elseif inputType == Enum.UserInputType.MouseButton5 then return "MB5"
	elseif inputType == Enum.UserInputType.Keyboard then return "Keyboard"
	else return "Unknown" end
end

function safeDestroy(object)
	if object and object.Parent then
		object:Destroy()
	end
end

local HttpService = game:GetService("HttpService")
local CONFIG_FOLDER = "AteotSettings"

local function getExecutorFunction(name)
	local direct = _G[name]
	if type(direct) == "function" then
		return direct
	end
	if type(getgenv) == "function" then
		local env = getgenv()
		if type(env[name]) == "function" then
			return env[name]
		end
	end
	return nil
end

local FS = {
	isfolder = getExecutorFunction("isfolder"),
	makefolder = getExecutorFunction("makefolder"),
	isfile = getExecutorFunction("isfile"),
	writefile = getExecutorFunction("writefile"),
	readfile = getExecutorFunction("readfile"),
	listfiles = getExecutorFunction("listfiles"),
}

local function refreshFS()
	FS.isfolder = getExecutorFunction("isfolder")
	FS.makefolder = getExecutorFunction("makefolder")
	FS.isfile = getExecutorFunction("isfile")
	FS.writefile = getExecutorFunction("writefile")
	FS.readfile = getExecutorFunction("readfile")
	FS.listfiles = getExecutorFunction("listfiles")
end

local function configFileSystemAvailable()
	refreshFS()
	return type(FS.makefolder) == "function"
		and type(FS.isfolder) == "function"
		and type(FS.writefile) == "function"
		and type(FS.readfile) == "function"
		and type(FS.listfiles) == "function"
end

local function ensureConfigFolder()
	refreshFS()
	if not configFileSystemAvailable() then
		return false, "Solara: file API unavailable"
	end

	local folderOK, exists = pcall(FS.isfolder, CONFIG_FOLDER)
	if folderOK and exists == true then
		return true, CONFIG_FOLDER
	end

	local createOK, createErr = pcall(FS.makefolder, CONFIG_FOLDER)
	if not createOK then
		local checkOK, nowExists = pcall(FS.isfolder, CONFIG_FOLDER)
		if not checkOK or not nowExists then
			return false, "Solara: cannot create " .. CONFIG_FOLDER .. ": " .. tostring(createErr)
		end
	end

	local checkOK, nowExists = pcall(FS.isfolder, CONFIG_FOLDER)
	if not checkOK or not nowExists then
		return false, "Solara: folder " .. CONFIG_FOLDER .. " was not created"
	end

	return true, CONFIG_FOLDER
end

local function sanitizeConfigName(name)
	name = tostring(name or "")
	name = name:gsub('[\\/:*?"<>|]', "_")
	name = name:gsub("%s+", " ")
	name = name:match("^%s*(.-)%s*$") or ""
	name = name:gsub("%.json$", "")
	if #name > 48 then
		name = name:sub(1, 48)
	end
	return name
end

local function configPath(name)
	return CONFIG_FOLDER .. "/" .. name .. ".json"
end

local function keyToString(key)
	if not key or key == Enum.KeyCode.Unknown then
		return "Unknown"
	end
	return key.Name
end

local function stringToKey(name)
	if type(name) ~= "string" or name == "Unknown" or name == "" then
		return Enum.KeyCode.Unknown
	end
	return Enum.KeyCode[name] or Enum.KeyCode.Unknown
end

local function saveConfig(configName)
	local folderOK, folderResult = ensureConfigFolder()
	if not folderOK then
		return false, folderResult
	end

	configName = sanitizeConfigName(configName)
	if configName == "" then
		return false, "Enter a config name"
	end

	local encodeOK, encoded = pcall(function()
		return HttpService:JSONEncode(buildConfigData())
	end)
	if not encodeOK then
		return false, "Solara: JSON encode failed: " .. tostring(encoded)
	end

	refreshFS()
	if type(FS.writefile) ~= "function" then
		return false, "Solara: writefile unavailable"
	end

	local path = configPath(configName)
	local writeOK, writeErr = pcall(FS.writefile, path, encoded)
	if not writeOK then
		return false, "Solara: save failed: " .. tostring(writeErr)
	end

	-- Проверяем, что файл реально появился.
	if type(FS.isfile) == "function" then
		local checkOK, exists = pcall(FS.isfile, path)
		if not checkOK or not exists then
			return false, "Solara: file was not created"
		end
	end

	return true, configName
end

local function getConfigFiles()
	local folderOK, folderResult = ensureConfigFolder()
	if not folderOK then
		return {}, folderResult
	end

	refreshFS()
	local listOK, result = pcall(FS.listfiles, CONFIG_FOLDER)
	if not listOK or type(result) ~= "table" then
		return {}, "Solara: listfiles failed"
	end

	local files = {}
	local seen = {}
	for _, path in ipairs(result) do
		local normalized = tostring(path):gsub("\\", "/")
		local name = normalized:match("([^/]+)%.json$")
		if name and not seen[name:lower()] then
			seen[name:lower()] = true
			table.insert(files, {name = name, path = tostring(path)})
		end
	end

	table.sort(files, function(a, b)
		return a.name:lower() < b.name:lower()
	end)
	return files
end

local function loadConfigFile(filePath)
	refreshFS()
	if type(FS.readfile) ~= "function" then
		return false, "Solara: readfile unavailable"
	end

	local readOK, raw = pcall(FS.readfile, filePath)
	if not readOK or type(raw) ~= "string" then
		return false, "Solara: could not read config"
	end

	local decodeOK, data = pcall(function()
		return HttpService:JSONDecode(raw)
	end)
	if not decodeOK or type(data) ~= "table" then
		return false, "Solara: invalid JSON config"
	end

	if type(data.Keys) == "table" then
		for name, value in pairs(data.Keys) do
			if Keys[name] ~= nil then
				Keys[name] = stringToKey(value)
			end
		end
	end

	if type(data.Features) == "table" then
		for name, value in pairs(data.Features) do
			if Features[name] ~= nil then
				Features[name] = value
			end
		end
	end

	if type(data.espSettings) == "table" then
		for name, value in pairs(data.espSettings) do
			if espSettings[name] ~= nil then
				espSettings[name] = value
			end
		end
	end

	if type(data.theme) == "string" and THEMES[data.theme] then
		currentThemeName = data.theme
	end

	return true, data
end

function makeDraggable(frame, handle)
	handle = handle or frame
	local dragging = false
	local dragInput, dragStart, startPos

	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position

			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	handle.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement then
			dragInput = input
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local delta = input.Position - dragStart
			frame.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset + delta.Y
			)
		end
	end)
end

function updateRainbowCursor()
	if not rainbowCursor or not Features.rainbowCursorEnabled then return end
	rainbowHue = (rainbowHue + 0.3) % 360
	local color = Color3.fromHSV(rainbowHue / 360, 1, 1)
	rainbowCursor.BackgroundColor3 = color
	rainbowCursor.Position = UDim2.new(0, mouse.X - 15, 0, mouse.Y - 15)
end

function toggleRainbowCursor(state)
	Features.rainbowCursorEnabled = state
	if state then
		if not rainbowCursorGui then
			createRainbowCursor()
		end
		rainbowCursorGui.Visible = true
		rainbowCursor.Visible = true
		mouse.Icon = "rbxasset://textures/ui/cursors/blank.cur"
	else
		if rainbowCursorGui then
			rainbowCursorGui.Visible = false
		end
		mouse.Icon = ""
	end
end

function animateParticles()
	for _, particle in ipairs(particleObjects) do
		if not particle.Parent then continue end
		local startPosition = particle.Position
		local newX = startPosition.X.Scale + math.random(-5, 5) / 100
		local newY = startPosition.Y.Scale - math.random(5, 15) / 100

		Tween(
			particle,
			math.random(30, 60) / 10,
			{
				Position = UDim2.fromScale(newX, newY),
				BackgroundTransparency = math.random(45, 85) / 100,
			},
			Enum.EasingStyle.Sine
		)
	end
end

	end
end)

		Text = text,
		TextColor3 = COLORS.SubText,
		TextSize = 12,
		Font = Enum.Font.GothamBold,
		TextXAlignment = Enum.TextXAlignment.Left,
		ZIndex = 10,
	})

	button.MouseEnter:Connect(function()
		if State.selectedButton ~= button then
			Tween(button, 0.15, { BackgroundColor3 = COLORS.PanelHover })
		end
	end)

	button.MouseLeave:Connect(function()
		if State.selectedButton ~= button then
			Tween(button, 0.15, { BackgroundColor3 = COLORS.Panel })
		end
	end)

	button.MouseButton1Click:Connect(function()
		if State.selectedButton and State.selectedButton ~= button then
			local oldIndicator = State.selectedButton:FindFirstChild("ActiveIndicator")
			local oldIcon = State.selectedButton:FindFirstChild("NavIcon")
			local oldText = State.selectedButton:FindFirstChild("NavText")

			Tween(State.selectedButton, 0.18, { BackgroundColor3 = COLORS.Panel })

			if oldText then Tween(oldText, 0.18, { TextColor3 = COLORS.SubText }) end

			currentPage.Visible = false
		end

	end)

	return button
end

function getMoveDirection(humanoid)
	local moveDir = humanoid.MoveDirection
	if moveDir.Magnitude == 0 then
		local cam = Workspace.CurrentCamera
		if cam then
			moveDir = cam.CFrame.LookVector
			moveDir = Vector3.new(moveDir.X, 0, moveDir.Z)
		end
	end
	return moveDir.Magnitude > 0 and moveDir.Unit or Vector3.zero
end

RunService.Heartbeat:Connect(function()
	if not Features.bhopEnabled then return end

	if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
		local character = player.Character
		if not character then return end

		local humanoid = character:FindFirstChildOfClass("Humanoid")
		local hrp = character:FindFirstChild("HumanoidRootPart")

		if humanoid and hrp and humanoid.Health > 0 then
			local moveDir = getMoveDirection(humanoid)

			if humanoid.FloorMaterial ~= Enum.Material.Air then
				humanoid.Jump = true

				if Features.bhopMode == "TP" and canTp and moveDir.Magnitude > 0 then
					canTp = false
					local jumpAirTime = 0.35
					local tpDistance = moveDir * (Features.bhopSpeed * jumpAirTime)
					hrp.CFrame = hrp.CFrame + tpDistance
					task.delay(0.15, function()
						canTp = true
					end)
				end
			end

			if Features.bhopMode == "NORMAL" and humanoid.FloorMaterial == Enum.Material.Air then
				if moveDir.Magnitude > 0 then
					local currentY = hrp.AssemblyLinearVelocity.Y
					hrp.AssemblyLinearVelocity = Vector3.new(
						moveDir.X * Features.bhopSpeed,
						currentY,
						moveDir.Z * Features.bhopSpeed
					)
				end
			end
		end
	end
end)

function triggerbotShoot(hit)
	if not hit then return end
	local character = hit.Parent
	if character and character:FindFirstChild("Humanoid") and character.Name ~= player.Name then
		local validPart = false
		
		if Features.triggerTargetPart == "All" then
			validPart = true
		elseif Features.triggerTargetPart == "Head" then
			if hit.Name == "Head" then validPart = true end
		elseif Features.triggerTargetPart == "Torso" then
			if hit.Name == "Torso" or hit.Name == "UpperTorso" or hit.Name == "LowerTorso" or hit.Name == "HumanoidRootPart" then
				validPart = true
			end
		end

		if validPart then
			if Features.triggerMode == "WithAim" then
				if triggerbotFlick then
					triggerbotFlick()
				end
			end
			mouse1press()
			task.wait()
			mouse1release()
		end
	end
end

triggerbotFlick = nil
do
	local flickActive = false
	local flickProgress = 0
	local flickStartCFrame = nil
	local flickTargetCFrame = nil
	local FLICK_DURATION = 0.15
	local FLICK_STRENGTH = 180

	local function getHumanizedOffset(progress)
		local smoothProgress = progress * progress * (3 - 2 * progress)
		local shakeX = math.sin(progress * math.pi * 30) * 0.3 * (1 - progress)
		local shakeY = math.cos(progress * math.pi * 25 + 1.2) * 0.3 * (1 - progress)
		local arcY = math.sin(progress * math.pi) * 1.5
		return smoothProgress, shakeX, shakeY, arcY
	end

	triggerbotFlick = function()
		if flickActive then return end
		if not camera then return end
		
		local randomSign = (math.random(1, 2) == 1 and 1) or -1
		local randomStrength = FLICK_STRENGTH * randomSign * (0.85 + math.random() * 0.3)
		
		local yaw = math.rad(randomStrength)
		local pitch = math.rad(math.random(-5, 5) * 0.5)
		
		local rotationY = CFrame.Angles(0, yaw, 0)
		local rotationX = CFrame.Angles(pitch, 0, 0)
		local totalRotation = rotationX * rotationY
		
		local currentPos = camera.CFrame.Position
		local newCFrame = CFrame.new(currentPos) * (camera.CFrame.Rotation * totalRotation)
		flickTargetCFrame = CFrame.new(currentPos, newCFrame.LookVector + currentPos)
		
		flickStartCFrame = camera.CFrame
		flickProgress = 0
		flickActive = true
	end

	RunService.RenderStepped:Connect(function(deltaTime)
		if not flickActive or not camera then
			if flickActive then flickActive = false end
			return
		end
		
		flickProgress = flickProgress + (deltaTime / FLICK_DURATION)
		
		if flickProgress >= 1 then
			flickProgress = 1
			if flickTargetCFrame then
				camera.CFrame = flickTargetCFrame
			end
			flickActive = false
			return
		end
		
		local smoothP, shakeX, shakeY, arcY = getHumanizedOffset(flickProgress)
		
		local startPos = flickStartCFrame.Position
		local startLook = flickStartCFrame.LookVector
		local startUp = flickStartCFrame.UpVector
		local startRight = flickStartCFrame.RightVector
		
		local targetLook = flickTargetCFrame.LookVector
		local targetUp = flickTargetCFrame.UpVector
		
		local currentLook = startLook:Lerp(targetLook, smoothP)
		local currentUp = startUp:Lerp(targetUp, smoothP)
		
		currentLook = currentLook.Unit
		currentUp = currentUp.Unit
		
		local shakeRight = startRight * math.rad(shakeX)
		local shakeUp = startUp * math.rad(shakeY)
		local shakeOffset = shakeRight + shakeUp
		
		local arcOffset = startUp * math.rad(arcY * 0.8)
		
		local noisyLook = (currentLook + shakeOffset + arcOffset).Unit
		
		local newCFrame = CFrame.lookAt(startPos, startPos + noisyLook, currentUp)
		
		camera.CFrame = newCFrame
	end)
end

RunService.RenderStepped:Connect(function()

		local hit = mouse.Target
		triggerbotShoot(hit)
	end
end)

--// =========================================================
--// ORIGINAL CAMERA TURN CODE
--// =========================================================

function startCameraTurn()
	local camera = workspace.CurrentCamera
	if not camera then return end

	local ACTIVATION_KEY = Enum.UserInputType.MouseButton1
	local TURN_DURATION = 0.15
	local TURN_STRENGTH = 180

	local isTurning = false
	local turnProgress = 0
	local startCFrame = nil
	local targetCFrame = nil

	local function getHumanizedOffset(progress)
		local smoothProgress = progress * progress * (3 - 2 * progress)
		local shakeX = math.sin(progress * math.pi * 30) * 0.3 * (1 - progress)
		local shakeY = math.cos(progress * math.pi * 25 + 1.2) * 0.3 * (1 - progress)
		local arcY = math.sin(progress * math.pi) * 1.5
		return smoothProgress, shakeX, shakeY, arcY
	end

	local function startTurn()
		if isTurning then return end
		if not camera then return end

		local randomSign = (math.random(1, 2) == 1 and 1) or -1
		local randomStrength = TURN_STRENGTH * randomSign * (0.85 + math.random() * 0.3)
		local currentLookVector = camera.CFrame.LookVector
		local currentUpVector = camera.CFrame.UpVector
		local yaw = math.rad(randomStrength)
		local pitch = math.rad(math.random(-5, 5) * 0.5)
		local rotationY = CFrame.Angles(0, yaw, 0)
		local rotationX = CFrame.Angles(pitch, 0, 0)
		local totalRotation = rotationX * rotationY
		local currentPos = camera.CFrame.Position
		local newCFrame = CFrame.new(currentPos) * (camera.CFrame.Rotation * totalRotation)
		targetCFrame = CFrame.new(currentPos, newCFrame.LookVector + currentPos)
		startCFrame = camera.CFrame
		turnProgress = 0
		isTurning = true
	end

	startTurn()

	local connection
	connection = RunService.RenderStepped:Connect(function(deltaTime)
		if not isTurning or not camera then
			if connection then connection:Disconnect() end
			if isTurning then isTurning = false end
			return
		end

		turnProgress = turnProgress + (deltaTime / TURN_DURATION)

		if turnProgress >= 1 then
			turnProgress = 1
			if targetCFrame then
				camera.CFrame = targetCFrame
			end
			isTurning = false
			connection:Disconnect()
			return
		end

		local smoothP, shakeX, shakeY, arcY = getHumanizedOffset(turnProgress)
		local startPos = startCFrame.Position
		local startLook = startCFrame.LookVector
		local startUp = startCFrame.UpVector
		local startRight = startCFrame.RightVector
		local targetLook = targetCFrame.LookVector
		local targetUp = targetCFrame.UpVector
		local currentLook = startLook:Lerp(targetLook, smoothP)
		local currentUp = startUp:Lerp(targetUp, smoothP)
		currentLook = currentLook.Unit
		currentUp = currentUp.Unit
		local shakeRight = startRight * math.rad(shakeX)
		local shakeUp = startUp * math.rad(shakeY)
		local shakeOffset = shakeRight + shakeUp
		local arcOffset = startUp * math.rad(arcY * 0.8)
		local noisyLook = (currentLook + shakeOffset + arcOffset).Unit
		local newCFrame = CFrame.lookAt(startPos, startPos + noisyLook, currentUp)
		camera.CFrame = newCFrame
	end)
end

function applySkyColor(color)
	Lighting.OutdoorAmbient = color
	Lighting.Ambient = color
	Lighting.FogColor = color
	Lighting.FogStart = 0
	Lighting.FogEnd = 2000
	if skyAtmosphere then
		skyAtmosphere.Color = color
	end
end

function startSkyMode()
	for _, v in pairs(Lighting:GetChildren()) do
		if v:IsA("Sky") then
			v:Destroy()
		end
	end

	skyAtmosphere = Lighting:FindFirstChildOfClass("Atmosphere")
	if not skyAtmosphere then
		skyAtmosphere = Instance.new("Atmosphere")
		skyAtmosphere.Parent = Lighting
	end
	skyAtmosphere.Density = 0.6
	skyAtmosphere.Haze = 10
	skyAtmosphere.Glare = 10
	skyAtmosphere.Decay = Color3.fromRGB(0, 0, 0)

	if Features.skyMode == "Rainbow" then
		skyConnection = RunService.RenderStepped:Connect(function()
			local hue = (tick() % 2) / 2
			local rainbowColor = Color3.fromHSV(hue, 1, 1)
			applySkyColor(rainbowColor)
		end)
	elseif Features.skyMode == "Red" then
		applySkyColor(Color3.fromRGB(255, 0, 0))
	elseif Features.skyMode == "Blue" then
		applySkyColor(Color3.fromRGB(0, 150, 255))
	end
end

function stopSkyMode()
	if skyConnection then
		skyConnection:Disconnect()
		skyConnection = nil
	end
	Lighting.OutdoorAmbient = Color3.fromRGB(0, 0, 0)
	Lighting.Ambient = Color3.fromRGB(0, 0, 0)
	Lighting.FogColor = Color3.fromRGB(0, 0, 0)
	Lighting.FogStart = 0
	Lighting.FogEnd = 100000
	if skyAtmosphere then
		skyAtmosphere.Color = Color3.fromRGB(0, 0, 0)
	end
end

RunService.RenderStepped:Connect(function()
	for plr, draw in pairs(espDrawings) do
		local char = plr.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local root = char and char:FindFirstChild("HumanoidRootPart")

		if Features.espEnabled and char and hum and root and hum.Health > 0 then
			local rootPos, onScreen = camera:WorldToViewportPoint(root.Position)
			if onScreen then
				local head = char:FindFirstChild("Head")
				local headPos = head and camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0)) or rootPos

				local sizeY = math.abs(headPos.Y - rootPos.Y) * 2
				sizeY = math.clamp(sizeY, 15, 500)
				local sizeX = sizeY * 0.65

				if espSettings.Boxes then
					draw.box.Size = UDim2.fromOffset(sizeX, sizeY)
					draw.box.Position = UDim2.fromOffset(rootPos.X - sizeX/2, rootPos.Y - sizeY/2)
					draw.box.Visible = true
				else
					draw.box.Visible = false
				end

				if espSettings.Tracers then
					local origin = Vector2.new(camera.ViewportSize.X / 2, camera.ViewportSize.Y)
					local target = Vector2.new(rootPos.X, rootPos.Y)
					local dist = (target - origin).Magnitude

					draw.tracer.Size = UDim2.fromOffset(1.5, dist)
					draw.tracer.Position = UDim2.fromOffset((origin.X + target.X)/2, (origin.Y + target.Y)/2)
					draw.tracer.Rotation = math.deg(math.atan2(target.Y - origin.Y, target.X - origin.X)) - 90
					draw.tracer.Visible = true
				else
					draw.tracer.Visible = false
				end

				if espSettings.ShowName then
					local text = plr.Name
					if espSettings.ShowDistance then
						local distMeters = math.floor((player.Character and player.Character:FindFirstChild("HumanoidRootPart") and (player.Character.HumanoidRootPart.Position - root.Position).Magnitude) or 0)
						text = text .. " [" .. distMeters .. "m]"
					end
					draw.nameTag.Text = text
					draw.nameTag.Position = UDim2.fromOffset(rootPos.X - 100, rootPos.Y - sizeY/2 - 18)
					draw.nameTag.Size = UDim2.fromOffset(200, 15)
					draw.nameTag.Visible = true
				else
					draw.nameTag.Visible = false
				end

				if espSettings.ShowHeldTool then
					local tool = char:FindFirstChildOfClass("Tool")
					draw.toolTag.Text = tool and tool.Name or "None"
					draw.toolTag.Position = UDim2.fromOffset(rootPos.X - 100, rootPos.Y + sizeY/2 + 2)
					draw.toolTag.Size = UDim2.fromOffset(200, 15)
					draw.toolTag.Visible = true
				else
					draw.toolTag.Visible = false
				end

				if espSettings.HealthBar then
					local healthPercent = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
					draw.hpBg.Size = UDim2.fromOffset(3, sizeY)
					draw.hpBg.Position = UDim2.fromOffset(rootPos.X - sizeX/2 - 6, rootPos.Y - sizeY/2)
					draw.hpBar.Size = UDim2.new(1, 0, healthPercent, 0)
					draw.hpBar.Position = UDim2.new(0, 0, 1 - healthPercent, 0)
					draw.hpBar.BackgroundColor3 = Color3.fromRGB(255 - (255 * healthPercent), 255 * healthPercent, 0)
					draw.hpBg.Visible = true
				else
					draw.hpBg.Visible = false
				end

			else
				draw.box.Visible = false
				draw.tracer.Visible = false
				draw.nameTag.Visible = false
				draw.toolTag.Visible = false
				draw.hpBg.Visible = false
			end
		else
			draw.box.Visible = false
			draw.tracer.Visible = false
			draw.nameTag.Visible = false
			draw.toolTag.Visible = false
			draw.hpBg.Visible = false
		end
	end
end)

		local row = New("TextLabel", {
			Parent = configListScroll,

--// =========================================================

function startAntiFling()
	local lastPosition = nil
	
	local steppedConn = RunService.Stepped:Connect(function()
		for _, plr in ipairs(Players:GetPlayers()) do
			if plr ~= player and plr.Character then
				local char = plr.Character
				for _, part in ipairs(char:GetDescendants()) do
					if part:IsA("BasePart") then
						part.CanCollide = false
						part.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
						part.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
						part.CustomPhysicalProperties = PhysicalProperties.new(0, 0, 0, 0, 0)
					end
				end
			end
		end
	end)
	table.insert(antiFlingConnections, steppedConn)

	local heartbeatConn = RunService.Heartbeat:Connect(function()
		pcall(function()
			local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
			if hrp then
				if hrp.AssemblyLinearVelocity.Magnitude > 200 or hrp.AssemblyAngularVelocity.Magnitude > 200 then
					hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
					hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
					if lastPosition then
						hrp.CFrame = lastPosition
					end
				elseif hrp.AssemblyLinearVelocity.Magnitude < 50 then
					lastPosition = hrp.CFrame
				end
			end
		end)
	end)
	table.insert(antiFlingConnections, heartbeatConn)
end

function stopAntiFling()
	for _, conn in ipairs(antiFlingConnections) do
		conn:Disconnect()
	end
	antiFlingConnections = {}
end

function applyTheme(color)
	COLORS.Accent = color
	openStroke.Color = color
	if Features.triggerbotEnabled then tbToggle.BackgroundColor3 = color end
	if Features.cursorUnlocked then cursorToggle.BackgroundColor3 = color end
	if Features.noclipEnabled then noclipToggle.BackgroundColor3 = color end
	if Features.espEnabled then espToggle.BackgroundColor3 = color end
	if Features.spinbotEnabled then spinbotToggle.BackgroundColor3 = color end
	if Features.bhopEnabled then bhopToggle.BackgroundColor3 = color end
	if Features.skyEnabled then skyToggle.BackgroundColor3 = color end
	if Features.stretchEnabled then stretchToggle.BackgroundColor3 = color end
	espRebindBtn.TextColor3 = color
	for _, obj in ipairs(themeObjects) do
		if obj:IsA("Frame") then
			if obj.Name == "ActiveIndicator" and obj.BackgroundTransparency == 1 then
				obj.BackgroundColor3 = color

			end
		elseif obj:IsA("TextLabel") then
			obj.TextColor3 = color
		end
	end
	updateKeybindsList()
	updatePreview()
end

function openMenu()
	if State.opened or State.animating then return end
	State.animating = true
	State.opened = true
	particleLayer.Visible = true
	main.Visible = true
	openButton.Visible = false
	main.Size = CLOSED_SIZE

	Tween(main, 0.45, { Size = MENU_SIZE }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
	task.delay(0.45, function() if State.opened then State.animating = false end end)
end

function closeMenu()
	if not State.opened or State.animating then return end
	State.animating = true
	State.opened = false

	Tween(main, 0.3, { Size = CLOSED_SIZE }, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
	task.delay(0.3, function()
		if not State.opened then
			main.Visible = false
			particleLayer.Visible = false
			openButton.Visible = true
		end
		State.animating = false
	end)
end

function toggleMenu()
	if State.opened then closeMenu() else openMenu() end
end

	if State.settingsOpen then return end
	State.settingsOpen = true
	settingsPopup.Visible = true
	settingsPopup.Size = UDim2.fromOffset(0, 0)

	Tween(settingsPopup, 0.25, { Size = UDim2.fromOffset(0, 0) })
	task.delay(0.25, function()
		if not State.settingsOpen and settingsPopup.Parent then settingsPopup.Visible = false end
	end)
end
