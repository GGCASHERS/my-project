local function getRoot()

    if char then
        return char:FindFirstChild("HumanoidRootPart")

-- ===== ПАРАМЕТРЫ =====
local active = false
local mode = "tp"
local flySpeed = 55
local tpSpeed = 70        -- снижено с 200: меньше шанс откидывания
local lagFactor = 0.12

-- ===== ССЫЛКИ ДЛЯ ОБХОДОВ =====
local hrpRef = nil
local blockReplication = false
local oldIndex, oldNamecall = nil, nil
local hookInstalled = false

-- ===== ОБХОД 1: Блокировка репликации =====
local function installReplicationBlock()
    if hookInstalled then return end
    local ok, err = pcall(function()
        local mt = getrawmetatable(game)
        oldIndex = mt.__index
        oldNamecall = mt.__namecall
        setreadonly(mt, false)
        mt.__index = newcclosure(function(self, key)
            if blockReplication and self == hrpRef
               and (key == "CFrame" or key == "Position" or key == "AssemblyLinearVelocity") then
                return rawget(self, key)
            end
            return oldIndex(self, key)
        end)
        mt.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            if blockReplication and self == hrpRef
               and (method == "SetNetworkOwner" or method == "SetPrimaryPartCFrame") then
                return nil
            end
            return oldNamecall(self, ...)
        end)
        setreadonly(mt, true)
    end)
    if ok then hookInstalled = true
    else warn("[Bypass1] fail: "..tostring(err)) end
end

-- ===== ОБХОД 2: Network Owner =====
local function grabNetworkOwner()
    pcall(function()
        for _, part in ipairs(lp.Character:GetDescendants()) do
            if part:IsA("BasePart") then
                pcall(function() part:SetNetworkOwner(lp) end)
            end
        end
    end)
end

-- ===== STATE ABUSE =====
local function abuseState()
    pcall(function()
        hum:SetStateEnabled(Enum.HumanoidStateType.Swimming, true)
        hum:SetStateEnabled(Enum.HumanoidStateType.Climbing, true)
        hum:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
        hum:SetStateEnabled(Enum.HumanoidStateType.GettingUp, false)
        hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
        hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
        hum:ChangeState(Enum.HumanoidStateType.Swimming)
    end)
end

local function restoreState()
    pcall(function()
        hum:SetStateEnabled(Enum.HumanoidStateType.Swimming, true)
        hum:SetStateEnabled(Enum.HumanoidStateType.Climbing, true)
        hum:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
        hum:SetStateEnabled(Enum.HumanoidStateType.GettingUp, true)
        hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true)
        hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
        hum:ChangeState(Enum.HumanoidStateType.GettingUp)
        hum.PlatformStand = false
    end)
end

-- ===== СОЕДИНЕНИЯ =====
local stateConn, moveConn, bg, bv = nil, nil, nil, nil

local function readMove()
    local move = Vector3.zero
    local camCF = camera.CFrame
    if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + camCF.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - camCF.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - camCF.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + camCF.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move = move + Vector3.new(0,1,0) end
    if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then move = move - Vector3.new(0,1,0) end
    return move, camCF
end

-- ===== FLY =====
local function startFly()
    bg = Instance.new("BodyGyro")
    bg.P = 9e4
    bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bg.CFrame = hrp.CFrame
    bg.Parent = hrp

    bv = Instance.new("BodyVelocity")
    bv.Velocity = Vector3.zero
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bv.P = 1250
    bv.Parent = hrp

    moveConn = RunService.RenderStepped:Connect(function(dt)
        if not active or mode ~= "fly" then return end
        local root = getRoot()
        if not root then return end
        local move, camCF = readMove()
        if move.Magnitude > 0 then move = move.Unit * flySpeed end
        bv.Velocity = bv.Velocity:Lerp(move, math.clamp(dt * 8, 0, 1))
        bg.CFrame = camCF
    end)
end

local function stopFly()
    if bg then bg:Destroy() bg = nil end
    if bv then bv:Destroy() bv = nil end
end

-- ===== TP С ОБХОДАМИ =====
local function startTP()
    moveConn = RunService.RenderStepped:Connect(function(dt)
        if not active or mode ~= "tp" then return end
        local char = lp.Character
        if not char then return end
        local root = getRoot()
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if not root or not humanoid then return end

        humanoid.PlatformStand = true

        local move, camCF = readMove()

        -- Обход 5: Velocity Clamp — держим скорость в лимите
        local effSpeed = tpSpeed
        if bypasses[5].enabled and effSpeed > 80 then
            effSpeed = 80
        end

        local currentPos = root.Position
        local targetPos = currentPos + (move * effSpeed * dt)

        -- Обход 4: Micro Steps — разбиваем движение
        if bypasses[4].enabled then
            local steps = 4
            local stepVec = (targetPos - currentPos) / steps
            for i = 1, steps do
                local p = currentPos + stepVec * i
                local newPos = currentPos:Lerp(p, lagFactor)
                root.CFrame = CFrame.new(newPos, newPos + camCF.LookVector)
                currentPos = newPos
            end
        else
            local newPos = currentPos:Lerp(targetPos, lagFactor)
            root.CFrame = CFrame.new(newPos, newPos + camCF.LookVector)
        end

        root.AssemblyLinearVelocity = Vector3.zero
    end)
end

local function stopTP()
    local char = lp.Character
    if char then
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if humanoid then humanoid.PlatformStand = false end
        local root = getRoot()
        if root then root.AssemblyLinearVelocity = Vector3.zero end
    end
end

-- ===== СТАРТ/СТОП =====
local function startAll()
    hrp = getRoot()
    hum = lp.Character and lp.Character:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end

    hrpRef = hrp

    -- Обход 1
    if bypasses[1].enabled then
        blockReplication = true
        installReplicationBlock()
    end

    -- Обход 2
    if bypasses[2].enabled then
        grabNetworkOwner()
    end

    abuseState()

    -- Обход 3: State Toggle
    local stateToggle = false
    stateConn = RunService.Heartbeat:Connect(function()
        if not active or not hum then return end
        pcall(function()
            if bypasses[3].enabled then
                stateToggle = not stateToggle
                hum:ChangeState(stateToggle and Enum.HumanoidStateType.Swimming
                                            or Enum.HumanoidStateType.Freefall)
            else
                if hum:GetState() ~= Enum.HumanoidStateType.Swimming then
                    hum:ChangeState(Enum.HumanoidStateType.Swimming)
                end
            end
        end)
    end)

    if mode == "fly" then startFly() else startTP() end
end

local function stopAll()
    blockReplication = false
    hrpRef = nil
    if stateConn then stateConn:Disconnect() stateConn = nil end
    if moveConn then moveConn:Disconnect() moveConn = nil end
    stopFly()
    stopTP()
    restoreState()
end

-- ===== ЛОГИКА =====
local function toggleActive()
    active = not active
    if active then
        startAll()
        btnMain.Text = "STOP"
        btnMain.BackgroundColor3 = Color3.fromRGB(0, 110, 0)
    else
        stopAll()
        btnMain.Text = "STATE FLY"
        btnMain.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    end
end

local function switchMode()
    if mode == "tp" then
        mode = "fly"; btnMode.Text = "MODE: FLY"
    else
        mode = "tp"; btnMode.Text = "MODE: TP"
    end
    if active then stopAll(); task.wait(0.1); startAll() end
end

btnMain.MouseButton1Click:Connect(toggleActive)
btnMode.MouseButton1Click:Connect(switchMode)
btnBinds.MouseButton1Click:Connect(function()
    bindFrame.Visible = not bindFrame.Visible
end)
btnBypass.MouseButton1Click:Connect(function()
    bypassFrame.Visible = not bypassFrame.Visible
end)

-- ===== БИНДЫ =====
local function startBind(key)
    awaitingBind = key
    local btn = ({toggle = bindToggleBtn, mode = bindModeBtn, menu = bindMenuBtn, bypass = bindBypassBtn})[key]
    if btn then btn.Text = "..."; btn.BackgroundColor3 = Color3.fromRGB(120, 90, 0) end
end

bindToggleBtn.MouseButton1Click:Connect(function() startBind("toggle") end)
bindModeBtn.MouseButton1Click:Connect(function() startBind("mode") end)
bindMenuBtn.MouseButton1Click:Connect(function() startBind("menu") end)
bindBypassBtn.MouseButton1Click:Connect(function() startBind("bypass") end)

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if awaitingBind then
        if input.UserInputType == Enum.UserInputType.Keyboard then
            binds[awaitingBind] = input.KeyCode
            local key = awaitingBind
            awaitingBind = nil
            local btn = ({toggle = bindToggleBtn, mode = bindModeBtn, menu = bindMenuBtn, bypass = bindBypassBtn})[key]
            if btn then btn.Text = binds[key].Name; btn.BackgroundColor3 = Color3.fromRGB(50, 50, 50) end
        end
        return
    end
    if input.KeyCode == binds.toggle then toggleActive()
    elseif input.KeyCode == binds.mode then switchMode()
    elseif input.KeyCode == binds.menu then bindFrame.Visible = not bindFrame.Visible
    elseif input.KeyCode == binds.bypass then bypassFrame.Visible = not bypassFrame.Visible
    end
end)
